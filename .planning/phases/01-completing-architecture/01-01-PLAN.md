## 迁移规则

### Routes 文件结构 (迁移后)
```javascript
// 迁移后的 routes/xxx.js 应只包含：
const express = require('express');
const router = express.Router();

// 1. 参数验证 (使用 Joi/Zod)
const validateRequest = require('../../common/middleware/validate');
const { xxxSchema } = require('../../schemas/xxx');

// 2. 路由定义 (只调用 service，不含业务逻辑)
router.post('/', validateRequest(xxxSchema), async (req, res, next) => {
  try {
    const result = await xxxService.create(req.body);
    res.json({ success: true, data: result });
  } catch (error) {
    next(error);
  }
});

// 3. 错误处理中间件
router.use(errorHandler);

module.exports = router;
```

## <action>

### 步骤 1: 创建统一验证中间件
```javascript
// common/middleware/validate.js
const Joi = require('joi');

const validate = (schema) => async (req, res, next) => {
  const { error, value } = schema.validate(req.body);
  if (error) {
    return next(new ValidationError(error.details[0].message));
  }
  req.validatedBody = value;
  next();
};

module.exports = validate;
```

### 步骤 2: 为每个路由文件创建 Schema
```javascript
// schemas/hitl.js
const Joi = require('joi');

const requestSchema = Joi.object({
  action: Joi.string().valid('approve', 'reject', 'cancel').required(),
  requestId: Joi.string().required(),
  reason: Joi.string().when('action', { is: 'reject', then: Joi.required() })
});

module.exports = { requestSchema };
```
