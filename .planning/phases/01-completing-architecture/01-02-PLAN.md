### 步骤 1: VectorSearchChannel._mockEmbed() 替换
```javascript
// BEFORE: domain/search/channels/VectorSearchChannel.js
_mockEmbed(text) {
  // 基于文本哈希生成假向量 - 这是错误的行为
  const hash = this._hashText(text);
  return Array.from({ length: this.dimension }, (_, i) =>
    (hash[(i * 7) % hash.length] * (i + 1)) / 1000
  );
}

// AFTER: 使用真实 Ollama 或 MiniMax embedding API
async _getEmbedding(text) {
  // 如果配置了 Ollama，使用 Ollama
  if (process.env.OLLAMA_BASE_URL) {
    const response = await fetch(`${process.env.OLLAMA_BASE_URL}/api/embeddings`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        model: process.env.OLLAMA_EMBEDDING_MODEL || 'mxbai-embed-large',
        prompt: text
      })
    });
    const data = await response.json();
    return data.embedding;
  }

  // 如果配置了 MiniMax，使用 MiniMax
  if (process.env.MINIMAX_API_KEY) {
    const response = await fetch('https://api.minimaxi.com/anthropic/v1/embeddings', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${process.env.MINIMAX_API_KEY}`
      },
      body: JSON.stringify({
        model: 'embedding-01',
        input: text
      })
    });
    const data = await response.json();
    return data.embedding;
  }

  // 严格模式：没有embedding服务时抛出错误
  throw new Error('Embedding service not configured. Set OLLAMA_BASE_URL or MINIMAX_API_KEY');
}
```

### 步骤 4: MCPToolRegistry.js mock 替换
```javascript
// BEFORE: 模拟 MCP 工具发现
async discoverTools() {
  return [
    { name: 'mock_tool', description: 'mock' }
  ];
}

// AFTER: 使用真实 MCP 协议
async discoverTools() {
  const mcpServers = this.config.mcpServers || [];
  const discoveredTools = [];

  for (const server of mcpServers) {
    try {
      const response = await fetch(`${server.url}/tools/list`, {
        method: 'GET',
        headers: { 'Content-Type': 'application/json' }
      });
      const data = await response.json();
      discoveredTools.push(...data.tools);
    } catch (error) {
      this.logger.warn(`MCP tool discovery failed for ${server.name}`, { error: error.message });
    }
  }

  return discoveredTools;
}
```
