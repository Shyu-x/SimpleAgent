## AgentLogger 规范定义

### 日志级别
| 级别 | 使用场景 | 输出目标 |
|------|---------|---------|
| DEBUG | 调试信息，开发环境 | 仅控制台 |
| INFO | 一般信息，操作记录 | 控制台 + 文件 |
| WARN | 警告，不影响功能 | 控制台 + 文件 |
| ERROR | 错误，影响功能 | 控制台 + 文件 + 告警 |
| FATAL | 致命错误，系统崩溃 | 控制台 + 文件 + 告警 + 通知 |

### 日志格式 (JSON)
```json
{
  "timestamp": "2026-04-26T12:00:00.000Z",
  "level": "INFO",
  "service": "ChatOrchestrator",
  "traceId": "abc123",
  "message": "处理用户消息",
  "context": {
    "userId": "user_001",
    "messageId": "msg_123",
    "intent": "weather_query"
  },
  "duration": 150
}
```

### 步骤 2: 创建日志转换脚本
```javascript
// scripts/convert-console-to-logger.js
const fs = require('fs');
const path = require('path');
const glob = require('glob');

const loggerTemplate = `const AgentLogger = require('../../infra/logger/AgentLogger');
const logger = new AgentLogger('% SERVICE_NAME %');

`;

const levelMap = {
  'console.log': 'logger.info',
  'console.debug': 'logger.debug',
  'console.warn': 'logger.warn',
  'console.error': 'logger.error'
};

glob.sync('backend/src/**/*.js').forEach(file => {
  let content = fs.readFileSync(file, 'utf-8');
  const relPath = path.relative('backend/src', file);

  // 检查是否已导入 logger
  if (!content.includes('AgentLogger')) {
    content = content.replace(
      /^(const|let|var)\s+\w+\s+=\s+require\(['"]@/?/,
      loggerTemplate.replace('% SERVICE_NAME %', relPath.split('/')[0]) + '$&'
    );
  }

  // 替换 console.* 为 logger.*
  Object.entries(levelMap).forEach(([consoleMethod, loggerMethod]) => {
    const regex = new RegExp(`console\\.${consoleMethod.split('.')[1]}\\s*\\(([^)]+)\\)`, 'g');
    content = content.replace(regex, (match, args) => {
      return `${loggerMethod}(${args})`;
    });
  });

  fs.writeFileSync(file, content);
});
```

### 步骤 3: 分类处理 console 日志

#### 类型 A: 替换为 logger（大多数情况）
```javascript
// BEFORE
console.log('用户发送消息:', userId);
console.error('API 调用失败:', error);

// AFTER
logger.info('用户发送消息', { userId });
logger.error('API 调用失败', { error: error.message });
```

#### 类型 B: 移除不必要的日志
```javascript
// BEFORE
console.log('进入函数');
console.log('循环次数:', i);
console.log('变量值:', variable);

// AFTER
// 移除（这些是调试日志，不需要）
```

#### 类型 C: 保留但包装（必须保留的 console）
```javascript
// BEFORE
console.error('UNCAUGHT EXCEPTION:', err);

// AFTER
// 在最外层未捕获异常处理器中保留
process.on('uncaughtException', (err) => {
  console.error('UNCAUGHT EXCEPTION:', err); // 这是允许的
  process.exit(1);
});
```

### 步骤 4: 确保 AgentLogger 完整实现
```javascript
// backend/src/infra/logger/AgentLogger.js
class AgentLogger {
  constructor(serviceName) {
    this.serviceName = serviceName;
    this.isProduction = process.env.NODE_ENV === 'production';
  }

  _format(level, message, context = {}) {
    return JSON.stringify({
      timestamp: new Date().toISOString(),
      level,
      service: this.serviceName,
      traceId: process.env.TRACE_ID || 'N/A',
      message,
      context,
      pid: process.pid
    });
  }

  debug(message, context = {}) {
    if (!this.isProduction) {
      console.log(this._format('DEBUG', message, context));
    }
  }

  info(message, context = {}) {
    console.log(this._format('INFO', message, context));
  }

  warn(message, context = {}) {
    console.warn(this._format('WARN', message, context));
  }

  error(message, context = {}) {
    console.error(this._format('ERROR', message, context));
  }

  fatal(message, context = {}) {
    console.error(this._format('FATAL', message, context));
  }

  // 专用方法
  logRequest(operation, params) {
    this.info(`${operation} 请求`, { params });
  }

  logResponse(operation, duration, result) {
    this.info(`${operation} 响应`, { duration, resultSize: JSON.stringify(result).length });
  }

  logToolExecution(toolName, args, success, result) {
    this.info(`工具执行: ${toolName}`, { args, success, result: success ? 'OK' : 'FAIL' });
  }

  logError(operation, error) {
    this.error(`${operation} 错误`, {
      error: error.message,
      stack: error.stack
    });
  }
}

module.exports = AgentLogger;
```
