```javascript
// backend/tests/unit/infra/CircuitBreaker.test.js
const CircuitBreaker = require('../../../src/infra/circuitBreaker/CircuitBreaker');
const CircuitState = require('../../../src/infra/circuitBreaker/CircuitState');

describe('CircuitBreaker', () => {
  let circuitBreaker;

  beforeEach(() => {
    circuitBreaker = new CircuitBreaker({
      name: 'test-service',
      failureThreshold: 3,
      successThreshold: 2,
      timeout: 1000
    });
  });

  describe('状态机转换', () => {
    test('初始状态为 CLOSED', () => {
      expect(circuitBreaker.state).toBe(CircuitState.CLOSED);
    });

    test('失败次数达到阈值时转换为 OPEN', async () => {
      const failingFn = jest.fn().mockRejectedValue(new Error('fail'));

      for (let i = 0; i < 3; i++) {
        await expect(circuitBreaker.execute(failingFn)).rejects.toThrow('fail');
      }

      expect(circuitBreaker.state).toBe(CircuitState.OPEN);
    });

    test('OPEN 状态下立即拒绝请求', async () => {
      circuitBreaker.state = CircuitState.OPEN;
      const fn = jest.fn();

      await expect(circuitBreaker.execute(fn)).rejects.toThrow('Circuit breaker is OPEN');
      expect(fn).not.toHaveBeenCalled();
    });

    test('半开状态下成功调用转换为 CLOSED', async () => {
      circuitBreaker.state = CircuitState.HALF_OPEN;
      const fn = jest.fn().mockResolvedValue('success');

      await expect(circuitBreaker.execute(fn)).resolves.toBe('success');
      expect(circuitBreaker.state).toBe(CircuitState.CLOSED);
    });
  });

  describe('熔断指标', () => {
    test('记录成功和失败调用', async () => {
      const successFn = jest.fn().mockResolvedValue('ok');
      const failFn = jest.fn().mockRejectedValue(new Error('fail'));

      await circuitBreaker.execute(successFn);
      await circuitBreaker.execute(successFn);
      await expect(circuitBreaker.execute(failFn)).rejects.toThrow();

      const metrics = circuitBreaker.getMetrics();
      expect(metrics.successes).toBe(2);
      expect(metrics.failures).toBe(1);
    });

    test('计算错误率', async () => {
      const failFn = jest.fn().mockRejectedValue(new Error('fail'));

      for (let i = 0; i < 10; i++) {
        try { await circuitBreaker.execute(failFn); } catch {}
      }

      const metrics = circuitBreaker.getMetrics();
      expect(metrics.errorRate).toBe(1);
    });
  });

  describe('事件钩子', () => {
    test('状态变化时触发回调', () => {
      const onStateChange = jest.fn();
      circuitBreaker.onStateChange(onStateChange);

      circuitBreaker.state = CircuitState.OPEN;
      expect(onStateChange).toHaveBeenCalledWith(CircuitState.OPEN, CircuitState.CLOSED);
    });
  });
});
```

```javascript
// backend/tests/unit/infra/QueueRateLimiter.test.js
const QueueRateLimiter = require('../../../src/infra/rateLimiter/QueueRateLimiter');

describe('QueueRateLimiter', () => {
  let rateLimiter;

  beforeEach(() => {
    rateLimiter = new QueueRateLimiter({
      maxQueueSize: 10,
      windowMs: 1000,
      maxRequests: 5
    });
  });

  afterEach(() => {
    rateLimiter.shutdown();
  });

  describe('请求限流', () => {
    test('允许在限制内的请求', async () => {
      for (let i = 0; i < 5; i++) {
        const result = await rateLimiter.acquire('user1');
        expect(result.allowed).toBe(true);
      }
    });

    test('超过限制后拒绝请求', async () => {
      for (let i = 0; i < 5; i++) {
        await rateLimiter.acquire('user1');
      }

      const result = await rateLimiter.acquire('user1');
      expect(result.allowed).toBe(false);
      expect(result.reason).toBe('Rate limit exceeded');
    });

    test('不同用户独立限流', async () => {
      for (let i = 0; i < 5; i++) {
        await rateLimiter.acquire('user1');
      }

      const result = await rateLimiter.acquire('user2');
      expect(result.allowed).toBe(true);
    });
  });

  describe('队列管理', () => {
    test('队列满时拒绝新请求', async () => {
      const limiter = new QueueRateLimiter({
        maxQueueSize: 2,
        windowMs: 1000,
        maxRequests: 1
      });

      await limiter.acquire('user1');
      await limiter.acquire('user2');

      const result = await limiter.acquire('user3');
      expect(result.allowed).toBe(false);
      expect(result.reason).toBe('Queue full');

      limiter.shutdown();
    });

    test('FIFO 顺序处理', async () => {
      const results = [];
      const limiter = new QueueRateLimiter({
        maxQueueSize: 10,
        windowMs: 100,
        maxRequests: 1
      });

      // 清空初始窗口
      await limiter.acquire('init');

      // 添加多个请求
      limiter.acquire('user1').then(r => results.push('user1'));
      limiter.acquire('user2').then(r => results.push('user2'));
      limiter.acquire('user3').then(r => results.push('user3'));

      // 等待处理
      await new Promise(resolve => setTimeout(resolve, 200));

      expect(results).toEqual(['user1', 'user2', 'user3']);

      limiter.shutdown();
    });
  });

  describe('指标采集', () => {
    test('返回当前指标', async () => {
      await rateLimiter.acquire('user1');
      await rateLimiter.acquire('user2');

      const metrics = rateLimiter.getMetrics();
      expect(metrics.queueSize).toBeDefined();
      expect(metrics.activeRequests).toBeDefined();
      expect(metrics.waitTime).toBeDefined();
    });
  });
});
```

```javascript
// backend/tests/unit/infra/ConfigCenter.test.js
const ConfigCenter = require('../../../src/infra/config/ConfigCenter');
const fs = require('fs');
const path = require('path');

describe('ConfigCenter', () => {
  const testConfigPath = path.join(__dirname, 'test-config.json');

  beforeEach(() => {
    // 重置单例
    ConfigCenter['_instance'] = null;
    if (fs.existsSync(testConfigPath)) {
      fs.unlinkSync(testConfigPath);
    }
  });

  afterEach(() => {
    ConfigCenter['_instance'] = null;
    if (fs.existsSync(testConfigPath)) {
      fs.unlinkSync(testConfigPath);
    }
  });

  describe('配置读写', () => {
    test('设置和获取配置', () => {
      const config = ConfigCenter.getInstance({ configPath: testConfigPath });

      config.set('key1', 'value1');
      expect(config.get('key1')).toBe('value1');

      config.set('nested.key', 'value2');
      expect(config.get('nested.key')).toBe('value2');
    });

    test('默认值', () => {
      const config = ConfigCenter.getInstance({ configPath: testConfigPath });

      expect(config.get('nonexistent', 'default')).toBe('default');
    });

    test('配置持久化', () => {
      const config1 = ConfigCenter.getInstance({ configPath: testConfigPath });
      config1.set('persistent', 'value');

      const config2 = ConfigCenter.getInstance({ configPath: testConfigPath });
      expect(config2.get('persistent')).toBe('value');
    });
  });

  describe('热更新', () => {
    test('监听配置变化', (done) => {
      const config = ConfigCenter.getInstance({ configPath: testConfigPath });
      config.set('watched', 'initial');

      config.on('change', (key, newValue, oldValue) => {
        if (key === 'watched') {
          expect(newValue).toBe('updated');
          expect(oldValue).toBe('initial');
          done();
        }
      });

      config.set('watched', 'updated');
    });

    test('重新加载配置', () => {
      const config = ConfigCenter.getInstance({ configPath: testConfigPath });
      config.set('reload', 'value1');

      // 外部修改文件
      const currentConfig = JSON.parse(fs.readFileSync(testConfigPath, 'utf-8'));
      currentConfig.reload = 'value2';
      fs.writeFileSync(testConfigPath, JSON.stringify(currentConfig));

      config.reload();
      expect(config.get('reload')).toBe('value2');
    });
  });
});
```

```javascript
// backend/tests/unit/infra/MetricsCollector.test.js
const MetricsCollector = require('../../../src/infra/metrics/MetricsCollector');

describe('MetricsCollector', () => {
  let metrics;

  beforeEach(() => {
    metrics = new MetricsCollector();
  });

  afterEach(() => {
    metrics.shutdown();
  });

  describe('Counter 指标', () => {
    test('递增计数器', () => {
      metrics.increment('requests_total');
      metrics.increment('requests_total');
      metrics.increment('requests_total');

      expect(metrics.get('requests_total')).toBe(3);
    });

    test('带标签的计数器', () => {
      metrics.increment('requests_total', { method: 'GET', status: '200' });
      metrics.increment('requests_total', { method: 'GET', status: '200' });
      metrics.increment('requests_total', { method: 'POST', status: '201' });

      expect(metrics.get('requests_total', { method: 'GET', status: '200' })).toBe(2);
    });
  });

  describe('Histogram 指标', () => {
    test('记录持续时间', () => {
      metrics.recordDuration('request_duration', 100);
      metrics.recordDuration('request_duration', 200);
      metrics.recordDuration('request_duration', 300);

      const stats = metrics.getHistogramStats('request_duration');
      expect(stats.count).toBe(3);
      expect(stats.min).toBe(100);
      expect(stats.max).toBe(300);
      expect(stats.mean).toBe(200);
    });

    test('计算百分位数', () => {
      for (let i = 1; i <= 100; i++) {
        metrics.recordDuration('latency', i);
      }

      const p50 = metrics.getPercentile('latency', 50);
      const p99 = metrics.getPercentile('latency', 99);

      expect(p50).toBe(50);
      expect(p99).toBe(99);
    });
  });

  describe('Prometheus 格式导出', () => {
    test('生成 Prometheus 格式指标', () => {
      metrics.increment('http_requests_total', { method: 'GET' });
      metrics.recordDuration('http_request_duration_seconds', 0.5);

      const output = metrics.toPrometheusFormat();

      expect(output).toContain('http_requests_total{method="GET"} 1');
      expect(output).toContain('http_request_duration_seconds_count 1');
      expect(output).toContain('http_request_duration_seconds_sum 0.5');
    });
  });
});
```
