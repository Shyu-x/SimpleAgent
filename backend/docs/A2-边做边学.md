## 实战 1：添加新的 API 路由

### 1.1 创建路由文件
**文件**: `backend/src/routes/hello.js`

```javascript
const express = require('express');
const router = express.Router();

/**
 * GET /api/hello
 * 测试路由
 */
router.get('/', (req, res) => {
  res.json({
    message: 'Hello World!',
    timestamp: new Date().toISOString()
  });
});

/**
 * GET /api/hello/:name
 * 带参数的路由
 */
router.get('/:name', (req, res) => {
  const { name } = req.params;
  res.json({
    message: `Hello, ${name}!`,
    timestamp: new Date().toISOString()
  });
});

module.exports = router;
```

### 1.2 注册路由
**文件**: `backend/src/index.js` 或 `backend/src/routes/index.js`

```javascript

// 使用路由

```

### 1.3 验证
```bash
curl http://localhost:30000/api/hello
# {"message":"Hello World!","timestamp":"..."}

curl http://localhost:30000/api/hello/Alice
# {"message":"Hello, Alice!","timestamp":"..."}
```

## 实战 2：添加新的 Service

### 2.1 创建 Service
**文件**: `backend/src/services/helloService.js`

```javascript
/**
 * Hello Service - 演示分层架构
 * 职责：处理业务逻辑，不涉及 HTTP
 */
class HelloService {
  /**
   * 生成问候语
   * @param {string} name - 名称
   * @returns {object} 问候信息
   */
  greet(name) {
    const hour = new Date().getHours();
    let timeOfDay;

    if (hour < 12) {
      timeOfDay = '早上';
    } else if (hour < 18) {
      timeOfDay = '下午';
    } else {
      timeOfDay = '晚上';
    }

    return {
      message: `${timeOfDay}好, ${name}!`,
      timeOfDay,
      hour,
      timestamp: new Date().toISOString()
    };
  }

  /**
   * 批量问候
   * @param {string[]} names - 名称列表
   */
  greetAll(names) {
    return names.map(name => this.greet(name));
  }
}

// 导出单例
module.exports = new HelloService();
```

### 2.2 在路由中使用
**文件**: `backend/src/routes/hello.js`

```javascript
const express = require('express');
const router = express.Router();
const helloService = require('../services/helloService');

router.get('/:name', (req, res) => {
  const { name } = req.params;
  // 调用 Service 而不是直接处理业务逻辑
  const result = helloService.greet(name);
  res.json(result);
});

module.exports = router;
```

## 实战 3：添加新的工具

### 3.1 创建工具文件
**文件**: `backend/src/services/tools/CalculatorTool.js`

```javascript
/**
 * 计算器工具
 * 演示如何添加工具到 Agent 系统
 */
class CalculatorTool {
  constructor() {
    this.name = 'calculator';
    this.description = '执行数学计算';
    this.parameters = {
      type: 'object',
      properties: {
        expression: {
          type: 'string',
          description: '数学表达式，如 "2 + 2" 或 "10 * 5"'
        }
      },
      required: ['expression']
    };
  }

  /**
   * 执行计算
   * @param {object} args - 参数 { expression: "2 + 2" }
   * @returns {object} 结果
   */
  async execute(args) {
    const { expression } = args;

    try {
      // 安全评估：只允许基本数学运算
      const sanitized = expression.replace(/[^0-9+\-*/().]/g, '');

      // 使用 Function 而不是 eval（稍微安全一点）
      const result = new Function(`return ${sanitized}`)();

      return {
        success: true,
        expression,
        result,
        formatted: `${expression} = ${result}`
      };
    } catch (error) {
      return {
        success: false,
        expression,
        error: error.message
      };
    }
  }
}

module.exports = CalculatorTool;
```

### 3.2 注册工具
**文件**: `backend/src/services/tools/toolRegistry.js`

```javascript
const CalculatorTool = require('./CalculatorTool');

// 在工具注册中添加

  // ... 其他工具
  new CalculatorTool(),
];

```

## 实战 4：添加中间件

### 4.1 创建日志中间件
**文件**: `backend/src/middleware/logger.js`

```javascript
/**
 * 请求日志中间件
 */
function logger(req, res, next) {
  const start = Date.now();

  // 请求完成时记录
  res.on('finish', () => {
    const duration = Date.now() - start;
    console.log(
      `${req.method} ${req.path} - ${res.statusCode} - ${duration}ms`
    );
  });

  next();
}

module.exports = logger;
```

### 4.2 使用中间件
**文件**: `backend/src/index.js`

```javascript
const logger = require('./middleware/logger');

// 在所有路由之前注册
app.use(logger);
```

```javascript
// 在已有方法中添加 RAG 调用示例
async retrieveWithRAG(query) {
  // 1. 问题改写
  const rewrittenQuery = await this.queryRewriteService.rewrite(query);
```

## 常见问题排查

### 问题：路由不工作
1. 检查路由是否正确注册
2. 检查路由文件路径
3. 检查拼写错误

### 问题：Service 报错
1. 检查是否正确导出
2. 检查依赖是否正确注入
3. 检查循环依赖

### 问题：修改代码没生效
1. 确保使用 `npm run dev` 而不是 `npm start`
2. 重启服务
3. 清除 node_modules/.cache
