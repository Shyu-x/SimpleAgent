# 中间件设计 - 熔断器、限流器实现

## 核心问题
如何在 Express 中实现可复用的中间件？熔断器和限流器是如何工作的？

## Express 中间件模式

### 基础结构
```javascript
// middleware/logger.js
function logger(req, res, next) {
  // 1. 处理请求前的事情
  const start = Date.now();

  // 2. 调用 next() 继续处理
  next();

  // 3. 处理请求后的事情
  const duration = Date.now() - start;
  console.log(`${req.method} ${req.path} - ${duration}ms`);
}

module.exports = logger;
```

### 中间件链
```
请求 → middleware1 → middleware2 → middleware3 → Route Handler
                 ↓            ↓            ↓
              next()       next()       next()
```

### 错误处理中间件
```javascript
// 错误处理中间件（4个参数）
function errorHandler(err, req, res, next) {
  console.error('Error:', err.message);
  res.status(500).json({ error: err.message });
}

// 使用
app.use(errorHandler);
```

## 熔断器实现

### 设计思路
```
CLOSED (正常) → 失败超过阈值 → OPEN (熔断)
    ↑                              │
    │      探测成功                 │
    └─────── HALF_OPEN ←─── 熔断超时
```

### 完整实现
**文件**: `backend/src/infra/circuitBreaker/CircuitBreaker.js`

```javascript
/**
 * 熔断器 - 防止级联故障
 *
 * 三种状态：
 * - CLOSED: 正常，请求通过，失败计数
 * - OPEN: 熔断，请求直接拒绝
 * - HALF_OPEN: 半开，允许探测请求
 */
class CircuitBreaker {
  constructor(options = {}) {
    this.failureThreshold = options.failureThreshold || 5;  // 打开阈值
    this.resetTimeout = options.resetTimeout || 30000;      // 重置超时(ms)
    this.halfOpenMaxCalls = options.halfOpenMaxCalls || 3;  // 半开探测次数

    this.state = 'CLOSED';         // 当前状态
    this.failureCount = 0;          // 连续失败计数
    this.successCount = 0;         // 半开成功计数
    this.nextAttempt = 0;          // 下次尝试时间
  }

  /**
   * 执行被保护的函数
   * @param {Function} fn - 要执行的异步函数
   * @returns {Promise} 执行结果
   */
  async execute(fn) {
    // 检查是否可以执行
    if (this.state === 'OPEN') {
      if (Date.now() < this.nextAttempt) {
        throw new CircuitOpenError('Circuit is OPEN');
      }
      // 超时了，转到 HALF_OPEN
      this.state = 'HALF_OPEN';
      this.successCount = 0;
    }

    // HALF_OPEN 状态限制并发
    if (this.state === 'HALF_OPEN') {
      if (this.successCount >= this.halfOpenMaxCalls) {
        throw new CircuitOpenError('Circuit is HALF_OPEN, max calls reached');
      }
    }

    try {
      const result = await fn();
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure();
      throw error;
    }
  }

  /**
   * 记录成功
   */
  onSuccess() {
    if (this.state === 'HALF_OPEN') {
      this.successCount++;
      if (this.successCount >= this.halfOpenMaxCalls) {
        // 连续成功，关闭熔断器
        this.state = 'CLOSED';
        this.failureCount = 0;
        console.log('Circuit Breaker: CLOSED (recovered)');
      }
    } else {
      // 正常状态成功，重置失败计数
      this.failureCount = 0;
    }
  }

  /**
   * 记录失败
   */
  onFailure() {
    this.failureCount++;

    if (this.state === 'HALF_OPEN') {
      // 半开状态失败，立即打开
      this.state = 'OPEN';
      this.nextAttempt = Date.now() + this.resetTimeout;
      console.log('Circuit Breaker: OPEN (half-open failed)');
    } else if (this.failureCount >= this.failureThreshold) {
      // 达到阈值，打开熔断器
      this.state = 'OPEN';
      this.nextAttempt = Date.now() + this.resetTimeout;
      console.log('Circuit Breaker: OPEN (threshold reached)');
    }
  }

  /**
   * 获取当前状态
   */
  getStatus() {
    return {
      state: this.state,
      failureCount: this.failureCount,
      nextAttempt: this.nextAttempt
    };
  }
}

/**
 * 熔断器打开错误
 */
class CircuitOpenError extends Error {
  constructor(message) {
    super(message);
    this.name = 'CircuitOpenError';
  }
}

module.exports = CircuitBreaker;
```

### 使用示例
```javascript
const CircuitBreaker = require('./infra/circuitBreaker/CircuitBreaker');

const breaker = new CircuitBreaker({
  failureThreshold: 3,
  resetTimeout: 30000
});

// 保护 API 调用
async function callMiniMaxAPI(messages) {
  return breaker.execute(() => miniMaxClient.chat(messages));
}

// Express 中间件包装
function withCircuitBreaker(fn, breaker) {
  return async (req, res, next) => {
    try {
      const result = await breaker.execute(() => fn(req, res));
      res.json(result);
    } catch (error) {
      if (error instanceof CircuitOpenError) {
        res.status(503).json({
          error: 'Service temporarily unavailable',
          retryAfter: 30
        });
      } else {
        next(error);
      }
    }
  };
}
```

## 限流器实现

### 队列限流器
```javascript
/**
 * 队列限流器 - 控制并发数
 */
class QueueRateLimiter {
  constructor(options = {}) {
    this.maxConcurrent = options.maxConcurrent || 10;  // 最大并发
    this.maxQueue = options.maxQueue || 100;           // 队列长度
    this.queueTimeout = options.queueTimeout || 30000; // 队列超时

    this.running = 0;        // 当前运行数
    this.queue = [];         // 等待队列
  }

  /**
   * 获取执行许可
   * @returns {Promise} resolve 表示获得许可，reject 表示被拒绝
   */
  async acquire() {
    // 如果有空闲槽位，立即执行
    if (this.running < this.maxConcurrent) {
      this.running++;
      return;
    }

    // 如果队列已满，拒绝
    if (this.queue.length >= this.maxQueue) {
      throw new RateLimitError('Queue is full');
    }

    // 加入队列等待
    return new Promise((resolve, reject) => {
      const timeout = setTimeout(() => {
        // 超时，从队列移除
        const index = this.queue.findIndex(item => item.resolve === resolve);
        if (index !== -1) {
          this.queue.splice(index, 1);
        }
        reject(new RateLimitError('Queue timeout'));
      }, this.queueTimeout);

      this.queue.push({ resolve, reject, timeout });
    });
  }

  /**
   * 释放执行许可
   */
  release() {
    this.running--;

    // 取出队列中的下一个
    if (this.queue.length > 0) {
      const next = this.queue.shift();
      clearTimeout(next.timeout);
      this.running++;
      next.resolve();
    }
  }

  /**
   * 获取状态
   */
  getStatus() {
    return {
      running: this.running,
      queueLength: this.queue.length,
      maxConcurrent: this.maxConcurrent
    };
  }
}

class RateLimitError extends Error {
  constructor(message) {
    super(message);
    this.name = 'RateLimitError';
  }
}

module.exports = QueueRateLimiter;
```

### Express 中间件包装
```javascript
const QueueRateLimiter = require('./infra/rateLimiter/QueueRateLimiter');

const limiter = new QueueRateLimiter({
  maxConcurrent: 50,
  maxQueue: 200
});

function rateLimitMiddleware(req, res, next) {
  limiter.acquire()
    .then(() => {
      // 获得许可，在响应结束后释放
      res.on('finish', () => limiter.release());
      next();
    })
    .catch(() => {
      res.status(429).json({
        error: 'Too many requests',
        retryAfter: 30
      });
    });
}

app.use('/api/', rateLimitMiddleware);
```

## 组合使用：熔断 + 限流

```javascript
// 创建带熔断的保护器
const breaker = new CircuitBreaker({ failureThreshold: 3 });

// 创建限流器
const limiter = new QueueRateLimiter({ maxConcurrent: 50 });

/**
 * 受保护的 API 调用
 */
async function protectedAPIcall(req, res) {
  // 1. 限流检查
  await limiter.acquire();

  try {
    // 2. 熔断器保护
    const result = await breaker.execute(() => callAPI(req.body));
    res.json(result);
  } catch (error) {
    if (error instanceof CircuitOpenError) {
      res.status(503).json({ error: 'Service unavailable' });
    } else {
      throw error;
    }
  } finally {
    limiter.release();
  }
}
```

## 熔断器模式 vs 限流器的区别

| 维度 | 熔断器 | 限流器 |
|------|--------|--------|
| **触发条件** | 下游故障（失败） | 流量过大（请求数） |
| **保护对象** | 下游服务 | 自身系统 |
| **行为** | 快速失败（不调用下游） | 排队或拒绝 |
| **恢复** | 自动探测恢复 | 请求减少自动恢复 |
| **类比** | 保险丝 | 红绿灯 |

## 新手常见问题

Q: 熔断器一直打开怎么办？
A: 检查下游服务是否真的挂了，可能是 `failureThreshold` 设置太低

Q: 限流导致请求失败怎么办？
A: 客户端应该实现重试机制（带指数退避）

Q: 两者可以不用吗？
A: 小项目可以不加，但生产环境必须使用以保证稳定性
