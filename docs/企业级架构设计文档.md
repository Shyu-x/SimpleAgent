## 一、架构设计理念

### 1.1 为什么需要分层架构

**问题**: 当前代码中业务逻辑散落在 `routes/` 目录，违反分层原则

```
当前问题:
routes/router.js (819行) ──→ 直接实例化 ModelRouter, HybridSearch, ModelPool
                              直接处理 SSE 流式响应
                              直接定义模型配置

企业级标准:
routes/router.js (应该 < 100行) ──→ 参数校验 ──→ 应用编排层 ──→ 领域服务层 ──→ 基础设施层
```

**分层价值**:
- 换模型供应商 → 只改 infra 层，业务代码 0 修改
- 换业务逻辑 → 只改 application/domain 层，基础设施 0 修改
- 换数据库 → 只改 infra 层，领域逻辑 0 修改

### 1.2 四层架构设计

```
┌─────────────────────────────────────────────────────────────────┐
│                        routes/ (接口层)                         │
│   职责: 参数校验、响应组装、错误捕获                              │
│   约束: 业务逻辑 0 行，每路由 < 100 行                           │
├─────────────────────────────────────────────────────────────────┤
```

### 3.2 检索协调器架构 (门面模式)

```
为什么需要门面模式:
- 封装检索复杂度：对上层隐藏多通道、后处理、结果合并逻辑
- 统一入口：调用方只需知道 SearchCoordinator，不知道内部有多少通道
- 便于扩展：新增通道/后处理器不影响调用方

设计实现:
```

```javascript
// domain/search/SearchCoordinator.js
/**
 * 检索协调器 - 门面类
 * 封装多通道并行执行和后处理流水线
 */
class SearchCoordinator {
  /**
   * @param {SearchChannel[]} channels - 检索通道列表
   * @param {SearchResultPostProcessor[]} postProcessors - 后处理器列表
   */
  constructor(channels, postProcessors) {
    this.channels = channels;
    this.postProcessors = postProcessors;
    this.defaultTopK = 10;
  }

  /**
   * 执行检索
   * @param {string} query - 查询文本
   * @param {SearchOptions} options - 检索选项
   * @returns {Promise<SearchResult[]>}
   */
  async search(query, options = {}) {
    const topK = options.topK || this.defaultTopK;
    const traceId = options.traceId;

    // 1. 并行执行所有启用的通道
    const channelResults = await this.executeChannels(query, options, traceId);

    // 2. 合并多通道结果
    let mergedResults = this.mergeChannelResults(channelResults);

    // 3. 后处理流水线
    for (const processor of this.postProcessors) {
      if (processor.enabled !== false) {
        mergedResults = await processor.process(mergedResults, query, options);
      }
    }

    // 4. 返回 topK 结果
    return mergedResults.slice(0, topK);
  }

  /**
   * 并行执行通道
   */
  async executeChannels(query, options, traceId) {
    const enabledChannels = this.channels.filter(ch => ch.enabled !== false);

    const results = await Promise.all(
      enabledChannels.map(async (channel) => {
        const startTime = Date.now();
        try {
          const channelResults = await channel.search(query, options);
          return {
            channel: channel.channelName,
            results: channelResults,
            duration: Date.now() - startTime,
            success: true
          };
        } catch (error) {
          logger.error(`Channel ${channel.channelName} failed`, { traceId, error: error.message });
          return {
            channel: channel.channelName,
            results: [],
            duration: Date.now() - startTime,
            success: false,
            error: error.message
          };
        }
      })
    );

    return results;
  }

  /**
   * 合并多通道结果
   */
  mergeChannelResults(channelResults) {
    const merged = new Map(); // content -> 合并后的结果

    for (const { channel, results } of channelResults) {
      for (const result of results) {
        const key = result.content;
        if (merged.has(key)) {
          // 已存在，取最高分并合并通道信息
          const existing = merged.get(key);
          existing.score = Math.max(existing.score, result.score);
          existing.channels.push(channel);
        } else {
          merged.set(key, {
            ...result,
            channels: [channel]
          });
        }
      }
    }

    return Array.from(merged.values());
  }
}
```

```javascript
// domain/search/postProcessors/DeduplicationProcessor.js
/**
 * 去重处理器
 * 移除内容相似度超过阈值的重复结果
 */
class DeduplicationProcessor extends PostProcessor {
  constructor(options = {}) {
    super(options);
    this.similarityThreshold = options.similarityThreshold || 0.85;
  }

  async process(results, query, options) {
    if (results.length <= 1) return results;

    const unique = [];
    for (const result of results) {
      const isDuplicate = unique.some(existing =>
        this.calculateSimilarity(result.content, existing.content) > this.similarityThreshold
      );
      if (!isDuplicate) {
        unique.push(result);
      }
    }

    return unique;
  }

  calculateSimilarity(text1, text2) {
    // 使用 Jaccard 相似度
    const tokens1 = new Set(text1.split(/\s+/));
    const tokens2 = new Set(text2.split(/\s+/));
    const intersection = new Set([...tokens1].filter(x => tokens2.has(x)));
    const union = new Set([...tokens1, ...tokens2]);
    return intersection.size / union.size;
  }
}
```

```javascript
// domain/search/postProcessors/RerankerProcessor.js
/**
 * 重排序处理器
 * 使用 LLM 评估相关性并重新排序
 */
class RerankerProcessor extends PostProcessor {
  constructor(options = {}) {
    super(options);
    this.llmClient = options.llmClient;
    this.topK = options.topK || 20;
  }

  async process(results, query, options) {
    if (results.length <= 1 || !this.llmClient) return results;

    // 限制重排数量
    const candidates = results.slice(0, this.topK);

    // 构建重排 prompt
    const prompt = this.buildRerankPrompt(query, candidates);

    // 调用 LLM 评估
    const response = await this.llmClient.chat([
      { role: 'user', content: prompt }
    ]);

    // 解析响应获取排序
    const rankings = this.parseRankings(response.content);

    // 根据排名重新排序
    const ranked = rankings.map((rank, index) => ({
      ...candidates[index],
      rerankScore: rank.score
    }));

    return ranked.sort((a, b) => b.rerankScore - a.rerankScore);
  }

  buildRerankPrompt(query, candidates) {
    return `请评估以下搜索结果与查询的相关性，并给出 1-10 的评分。

查询: ${query}

搜索结果:
${candidates.map((c, i) => `[${i + 1}] ${c.content}`).join('\n')}

请按以下 JSON 格式返回评分（只返回 JSON，不要其他内容）:
{"rankings": [{"index": 1, "score": 9.5}, ...]}`;
  }
}
```

### 3.4 文档入库 Pipeline (模板方法模式)

```
为什么需要模板方法模式:
- 文档入库流程固定：解析 -> 清洗 -> 分块 -> 向量化 -> 索引
- 每个节点实现不同，但执行骨架相同
- 便于监控：每个节点有独立的日志和错误处理

设计实现:
```

```javascript
// domain/rag/ingestion/IngestionPipeline.js
/**
 * 文档入库流水线
 * 使用模板方法模式定义执行骨架
 */
class IngestionPipeline {
  constructor(nodes = []) {
    this.nodes = nodes;
    this.logger = new PipelineLogger();
  }

  /**
   * 添加节点
   */
  addNode(node) {
    if (!(node instanceof IngestionNode)) {
      throw new Error('Must be instanceof IngestionNode');
    }
    this.nodes.push(node);
    this.nodes.sort((a, b) => a.order - b.order);
    return this;
  }

  /**
   * 执行流水线 - 模板方法
   */
  async process(document, options = {}) {
    const traceId = options.traceId || generateId();
    const context = {
      document,
      traceId,
      output: null,
      metadata: {
        startTime: Date.now(),
        nodeResults: []
      }
    };

    // 流水线执行
    for (const node of this.nodes) {
      // 检查是否应该执行
      if (!node.shouldExecute(context)) {
        this.logger.skip(node.nodeName, traceId);
        continue;
      }

      // 节点开始
      this.logger.nodeStart(node.nodeName, traceId);
      const nodeStartTime = Date.now();

      try {
        // 执行节点
        const result = await node.execute(context);

        // 转换上下文
        context[node.outputKey] = result;
        context.metadata.nodeResults.push({
          node: node.nodeName,
          success: true,
          duration: Date.now() - nodeStartTime
        });

        this.logger.nodeEnd(node.nodeName, traceId, true, Date.now() - nodeStartTime);
      } catch (error) {
        context.metadata.nodeResults.push({
          node: node.nodeName,
          success: false,
          duration: Date.now() - nodeStartTime,
          error: error.message
        });

        this.logger.nodeEnd(node.nodeName, traceId, false, Date.now() - nodeStartTime, error);

        // 根据节点错误处理策略决定是否继续
        if (!node.continueOnError) {
          throw error;
        }
      }
    }

    // 最终输出
    context.metadata.totalDuration = Date.now() - context.metadata.startTime;
    this.logger.pipelineEnd(traceId, context.metadata);

    return context;
  }
}
```

```javascript
/**
 * 智能分块节点
 * 支持多种分块策略
 */
class ChunkNode extends IngestionNode {
  constructor(options = {}) {
    super({ name: 'ChunkNode', order: 30, outputKey: 'chunks', ...options });
    this.strategy = options.strategy || 'semantic'; // 'semantic' | 'recursive' | 'fixed'
    this.chunkSize = options.chunkSize || 500;
    this.overlap = options.overlap || 50;
  }

  async execute(context) {
    const text = context.parsed?.text || context.document.content;

    switch (this.strategy) {
      case 'semantic':
        return this.semanticChunk(text, context);
      case 'recursive':
        return this.recursiveChunk(text);
      case 'fixed':
        return this.fixedChunk(text);
      default:
        return this.semanticChunk(text, context);
    }
  }

  /**
   * 语义分块 - 按语义段落分块
   */
  async semanticChunk(text, context) {
    // 使用 LLM 识别语义边界
    const prompt = `请将以下文本分割成语义完整的段落，每个段落作为一个独立的 chunk。

要求：
1. 段落应该有明确的主题
2. 保留段落之间的边界标记 [[CHUNK_BOUNDARY]]
3. 不要破坏代码块、表格等结构

文本：
${text}`;

    const response = await llmClient.chat([
      { role: 'user', content: prompt }
    ]);

    const chunks = response.content
      .split('[[CHUNK_BOUNDARY]]')
      .map(c => c.trim())
      .filter(c => c.length > 0);

    return chunks.map((content, index) => ({
      content,
      index,
      metadata: {
        strategy: 'semantic',
        documentId: context.document.id,
        charCount: content.length
      }
    }));
  }

  /**
   * 递归分块 - 按分隔符层级分块
   */
  recursiveChunk(text) {
    const separators = ['\n\n', '\n', '. ', ' '];
    const chunks = [];

    const split = (text, separatorIndex) => {
      if (separatorIndex >= separators.length || text.length <= this.chunkSize) {
        if (text.trim()) {
          chunks.push({
            content: text.trim(),
            metadata: { strategy: 'recursive', separator: separators[separatorIndex] }
          });
        }
        return;
      }

      const separator = separators[separatorIndex];
      const parts = text.split(separator);

      let currentChunk = '';
      for (const part of parts) {
        if (currentChunk.length + part.length > this.chunkSize) {
          if (currentChunk.trim()) {
            chunks.push({
              content: currentChunk.trim(),
              metadata: { strategy: 'recursive', separator }
            });
          }
          currentChunk = part + separator;
        } else {
          currentChunk += part + separator;
        }
      }

      if (currentChunk.trim()) {
        split(currentChunk, separatorIndex + 1);
      }
    };

    split(text, 0);
    return chunks.map((c, i) => ({ ...c, index: i }));
  }
}
```

```javascript
// infra/circuitBreaker/CircuitBreaker.js
/**
 * 三态熔断器
 * CLOSED -> OPEN -> HALF_OPEN -> CLOSED
 */
class CircuitBreaker {
  constructor(options = {}) {
    this.name = options.name || 'default';
    this.failureThreshold = options.failureThreshold || 5;      // 失败次数阈值
    this.successThreshold = options.successThreshold || 3;      // 成功次数阈值
    this.timeout = options.timeout || 30000;                   // 熔断持续时间
    this.halfOpenMaxCalls = options.halfOpenMaxCalls || 3;    // 半开状态最大放行数

    this.state = STATES.CLOSED;
    this.failureCount = 0;
    this.successCount = 0;
    this.nextAttempt = 0;
    this.halfOpenCalls = 0;

    this.logger = options.logger || console;
  }

  /**
   * 执行被保护的函数
   * @param {Function} fn - 要执行的函数
   * @param {Function} fallback - 降级函数
   * @returns {Promise<any>}
   */
  async execute(fn, fallback) {
    // 检查状态
    if (this.state === STATES.OPEN) {
      if (Date.now() < this.nextAttempt) {
        this.logger.debug(`Circuit ${this.name} OPEN, calling fallback`);
        return fallback ? fallback() : null;
      }
      // 超时，进入半开
      this.transitionTo(STATES.HALF_OPEN);
    }

    // 半开状态限流
    if (this.state === STATES.HALF_OPEN) {
      if (this.halfOpenCalls >= this.halfOpenMaxCalls) {
        return fallback ? fallback() : null;
      }
      this.halfOpenCalls++;
    }

    try {
      const result = await fn();
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure();
      return fallback ? fallback() : null;
    }
  }

  onSuccess() {
    if (this.state === STATES.HALF_OPEN) {
      this.successCount++;
      if (this.successCount >= this.successThreshold) {
        this.transitionTo(STATES.CLOSED);
      }
    } else {
      this.failureCount = 0;
    }
  }

  onFailure() {
    this.failureCount++;

    if (this.state === STATES.HALF_OPEN) {
      this.transitionTo(STATES.OPEN);
    } else if (this.failureCount >= this.failureThreshold) {
      this.transitionTo(STATES.OPEN);
    }
  }

  transitionTo(newState) {
    const oldState = this.state;
    this.state = newState;

    if (newState === STATES.OPEN) {
      this.nextAttempt = Date.now() + this.timeout;
    } else if (newState === STATES.CLOSED) {
      this.failureCount = 0;
      this.successCount = 0;
    } else if (newState === STATES.HALF_OPEN) {
      this.halfOpenCalls = 0;
      this.successCount = 0;
    }

    this.logger.info(`Circuit ${this.name}: ${oldState} -> ${newState}`);
  }

  getState() {
    return {
      name: this.name,
      state: this.state,
      failureCount: this.failureCount,
      nextAttempt: this.nextAttempt
    };
  }
}
```

```javascript
 * 熔断器工厂 - 工厂模式

class CircuitBreakerFactory {
  constructor() {
    this.breakers = new Map();
  }

  get(name, options) {
    if (!this.breakers.has(name)) {
      this.breakers.set(name, new CircuitBreaker({ name, ...options }));
    }
    return this.breakers.get(name);
  }

  getAll() {
    return Array.from(this.breakers.values()).map(b => b.getState());
  }

  reset(name) {
    const breaker = this.breakers.get(name);
    if (breaker) {
      breaker.transitionTo(STATES.CLOSED);
    }
  }

  resetAll() {
    for (const breaker of this.breakers.values()) {
      breaker.transitionTo(STATES.CLOSED);
    }
  }
}
```

```javascript
// domain/model/ModelRouter.js
/**
 * 模型路由器
 * 支持多候选模型、自动故障转移、首包探测
 */
class ModelRouter {
  constructor(clients, options = {}) {
    this.clients = clients;  // Map<modelId, ChatModelClient>
    this.circuitBreakers = new Map();
    this.probeBuffer = options.probeBuffer || new ProbeBufferingCallback();

    // 为每个客户端创建熔断器
    for (const [modelId, client] of clients) {
      this.circuitBreakers.set(modelId, breakerFactory.get(`model_${modelId}`));
    }
  }

  /**
   * 路由到最佳模型
   * @param {ChatRequest} request - 聊天请求
   * @param {string[]} candidates - 候选模型列表（按优先级排序）
   * @returns {Promise<{modelId: string, stream: AsyncIterator}>}
   */
  async route(request, candidates) {
    const errors = [];

    for (const modelId of candidates) {
      const client = this.clients.get(modelId);
      const breaker = this.circuitBreakers.get(modelId);

      if (!client) {
        errors.push({ modelId, error: 'Model not found' });
        continue;
      }

      try {
        // 使用熔断器执行
        const result = await breaker.execute(
          () => client.streamChat(request.messages, request.options),
          () => null  // 降级返回 null
        );

        if (result) {
          return {
            modelId,
            client,
            stream: this.probeBuffer.wrap(result.stream)
          };
        }
      } catch (error) {
        errors.push({ modelId, error: error.message });
      }
    }

    // 所有模型都失败
    throw new AppError(
      'ALL_MODELS_UNAVAILABLE',
      `所有候选模型均不可用: ${JSON.stringify(errors)}`
    );
  }

  /**
   * 添加候选模型
   */
  addCandidate(modelId, client, options = {}) {
    this.clients.set(modelId, client);
    this.circuitBreakers.set(modelId, breakerFactory.get(`model_${modelId}`, options));
  }

  /**
   * 获取模型状态
   */
  getModelStates() {
    const states = [];
    for (const [modelId, breaker] of this.circuitBreakers) {
      states.push({
        modelId,
        ...breaker.getState()
      });
    }
    return states;
  }
}
```

### 6.1 分层架构改造

> **为什么企业里需要它**: 避免业务逻辑散落在接口层，导致换技术方案时需要修改大量代码

> **它解决什么真实问题**: 之前换模型供应商需要改 routes/ 和 application/ 两层，分层后只需要改 infra/ 层

> **STAR 描述**:
> - Situation: 项目初期为了快速迭代，业务逻辑直接写在路由层
> - Task: 随着功能增加，代码耦合严重，换模型供应商需要改 5 个文件
> - Action: 重构为四层架构，抽取应用编排层，定义领域边界
> - Result: 换供应商只需改 1 个 infra/ 实现类，开发效率提升 60%

### 6.2 熔断器设计

> **为什么企业里需要它**: 生产环境中模型服务不稳定，没有熔断会导致级联失败

> **它解决什么真实问题**: 模型超时导致整个系统卡死，用户无法发送消息

> **STAR 描述**:
> - Situation: 某次模型 API 响应超时，线程池耗尽
> - Task: 设计熔断机制，当模型失败率达到阈值时自动降级
> - Action: 实现三态熔断器 (CLOSED/OPEN/HALF_OPEN)，配合降级链
> - Result: 模型故障时自动切换到备用模型，用户无感知，故障恢复时间从 5 分钟降到 10 秒

### 6.3 检索 Pipeline

> **为什么企业里需要它**: 文档入库流程复杂，需要可观测、可扩展的流水线

> **它解决什么真实问题**: 之前文档入库是一个大函数，出问题无法定位到具体步骤

> **STAR 描述**:
> - Situation: 知识库检索质量不佳，需要增强文档处理流程
