### 1.2 安全威胁矩阵

| 威胁类型 | 描述 | 防护措施 | 严重程度 |
|---------|------|---------|---------|
| SQL 注入 | 恶意 SQL 语句注入 | 参数化查询、输入验证 | 🔴 严重 |
| XSS | 跨站脚本攻击 | 输出编码、Content Security Policy | 🔴 严重 |
| CSRF | 跨站请求伪造 | CSRF Token、同源检测 | 🟠 高 |
| 认证绕过 | 未授权访问 | 强认证机制、Token 校验 | 🔴 严重 |
| 敏感信息泄露 | 密钥/Token 泄露 | 加密存储、环境变量分离 | 🔴 严重 |
| 速率滥用 | DDoS/暴力破解 | 限流、验证码 | 🟠 高 |
| 提示注入 | Prompt 注入攻击 | 输入过滤、沙箱隔离 | 🟡 中 |
| 模型投毒 | 恶意训练数据 | RAG 源验证、内容过滤 | 🟠 高 |

### 2.1 API Key 管理

```javascript
// middleware/auth/apiKeyAuth.js
/**
 * API Key 认证中间件
 */
async function apiKeyAuth(req, res, next) {
  const apiKey = req.headers['x-api-key'] || req.query.apiKey;

  if (!apiKey) {
    return res.status(401).json({
      success: false,
      code: '1006', // AUTH_API_KEY_MISSING
      message: 'API Key 缺失'
    });
  }

  // 验证 API Key 格式
  if (!isValidApiKeyFormat(apiKey)) {
    return res.status(401).json({
      success: false,
      code: '1004', // AUTH_TOKEN_FORMAT_ERROR
      message: 'API Key 格式错误'
    });
  }

  // 查找 API Key 对应的用户
  const user = await User.findByApiKey(hashApiKey(apiKey));

  if (!user) {
    return res.status(401).json({
      success: false,
      code: '1005', // AUTH_API_KEY_INVALID
      message: 'API Key 无效'
    });
  }

  if (user.status !== 'active') {
    return res.status(403).json({
      success: false,
      code: '1008', // AUTH_USER_DISABLED
      message: '用户已被禁用'
    });
  }

  // 附加用户信息到请求
  req.user = user;
  req.apiKeyId = user.apiKeyId;

  next();
}

// API Key 格式验证
function isValidApiKeyFormat(key) {
  // 格式: aichat_ + 32位随机字符串
  return /^aichat_[a-f0-9]{32}$/i.test(key);
}

// API Key 哈希存储
function hashApiKey(apiKey) {
  return crypto.createHash('sha256').update(apiKey).digest('hex');
}
```

### 2.2 Token 认证

```javascript
// middleware/auth/tokenAuth.js
const jwt = require('jsonwebtoken');

/**
 * JWT Token 认证中间件
 */
async function tokenAuth(req, res, next) {
  const authHeader = req.headers.authorization;

  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({
      success: false,
      code: '1001', // AUTH_TOKEN_MISSING
      message: 'Token 缺失'
    });
  }

  const token = authHeader.substring(7);

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET, {
      issuer: 'aichat',
      audience: 'aichat-api'
    });

    // 查找用户
    const user = await User.findById(decoded.userId);

    if (!user) {
      return res.status(401).json({
        success: false,
        code: '1009', // AUTH_USER_NOT_FOUND
        message: '用户不存在'
      });
    }

    if (user.status !== 'active') {
      return res.status(403).json({
        success: false,
        code: '1008', // AUTH_USER_DISABLED
        message: '用户已被禁用'
      });
    }

    // 检查 Token 是否过期
    if (decoded.exp < Date.now() / 1000) {
      return res.status(401).json({
        success: false,
        code: '1003', // AUTH_TOKEN_EXPIRED
        message: 'Token 已过期'
      });
    }

    req.user = user;
    req.tokenPayload = decoded;

    next();
  } catch (error) {
    if (error.name === 'TokenExpiredError') {
      return res.status(401).json({
        success: false,
        code: '1003',
        message: 'Token 已过期'
      });
    }

    return res.status(401).json({
      success: false,
      code: '1002', // AUTH_TOKEN_INVALID
      message: 'Token 无效'
    });
  }
}

/**
 * Token 生成
 */
function generateToken(user, options = {}) {
  const { expiresIn = '7d' } = options;

  return jwt.sign(
    {
      userId: user.id,
      email: user.email,
      role: user.role
    },
    process.env.JWT_SECRET,
    {
      issuer: 'aichat',
      audience: 'aichat-api',
      expiresIn
    }
  );
}
```

```javascript
  admin: [
    Permissions.ADMIN_ALL,
    Permissions.USER_READ,
    Permissions.USER_WRITE,
    Permissions.USER_DELETE,
    Permissions.CHAT_READ,
    Permissions.CHAT_WRITE,
    Permissions.CHAT_DELETE,
    Permissions.AGENT_READ,
    Permissions.AGENT_WRITE,
    Permissions.AGENT_EXECUTE,
    Permissions.KB_READ,
    Permissions.KB_WRITE,
    Permissions.KB_ADMIN
  ],
  user: [
    Permissions.CHAT_READ,
    Permissions.CHAT_WRITE,
    Permissions.AGENT_READ,
    Permissions.AGENT_EXECUTE,
    Permissions.KB_READ,
    Permissions.KB_WRITE
  ],
  guest: [
    Permissions.CHAT_READ
  ]
};

/**
 * 权限检查中间件
 */
function requirePermission(...requiredPermissions) {
  return async (req, res, next) => {
    const user = req.user;

    if (!user) {
      return res.status(401).json({
        success: false,
        code: '1001',
        message: '未认证'
      });
    }

    const userPermissions = Roles[user.role] || [];

    const hasPermission = requiredPermissions.every(permission =>
      userPermissions.includes(permission) || userPermissions.includes(Permissions.ADMIN_ALL)
    );

    if (!hasPermission) {
      return res.status(403).json({
        success: false,
        code: '1007', // AUTH_PERMISSION_DENIED
        message: '权限不足',
        required: requiredPermissions
      });
    }

    next();
  };
}

  userController.delete
```

```javascript
// middleware/validation/inputValidator.js
const Joi = require('joi');

/**
 * 通用输入验证中间件工厂
 */
function validate(schema) {
  return async (req, res, next) => {
    const dataToValidate = { ...req.body, ...req.query, ...req.params };

    const { error, value } = schema.validate(dataToValidate, {
      abortEarly: false,
      stripUnknown: true
    });

    if (error) {
      const errors = error.details.map(detail => ({
        field: detail.path.join('.'),
        message: detail.message,
        code: 'INVALID_FIELD'
      }));

      return res.status(400).json({
        success: false,
        code: '2001', // VALIDATION_REQUIRED_FIELD
        message: '参数校验失败',
        errors
      });
    }

    // 验证通过，替换为清理后的值
    req.body = value.body || {};
    req.query = value.query || {};

    next();
  };
}
```

### 3.2 XSS 防护

```javascript
// middleware/security/xssProtection.js
const DOMPurify = require('isomorphic-dompurify');

/**
 * XSS 防护中间件
 */
function xssProtection(req, res, next) {
  // 清理请求体
  if (req.body && typeof req.body === 'object') {
    req.body = sanitizeObject(req.body);
  }

  // 清理查询参数
  if (req.query && typeof req.query === 'object') {
    req.query = sanitizeObject(req.query);
  }

  next();
}

/**
 * 递归清理对象
 */
function sanitizeObject(obj) {
  if (typeof obj === 'string') {
    return DOMPurify.sanitize(obj, {
      ALLOWED_TAGS: ['b', 'i', 'em', 'strong', 'a', 'code', 'pre'],
      ALLOWED_ATTR: ['href', 'target', 'rel']
    });
  }

  if (Array.isArray(obj)) {
    return obj.map(item => sanitizeObject(item));
  }

  if (typeof obj === 'object' && obj !== null) {
    const sanitized = {};
    for (const [key, value] of Object.entries(obj)) {
      // 跳过某些字段不做清理（如代码内容）
      if (key === 'code' || key === 'codeBlock' || key === 'rawContent') {
        sanitized[key] = value;
      } else {
        sanitized[key] = sanitizeObject(value);
      }
    }
    return sanitized;
  }

  return obj;
}
```

### 3.3 Prompt 注入防护

```javascript
// services/security/promptGuard.js
/**
 * Prompt 注入检测
 */
class PromptGuard {
  constructor(options = {}) {
    this.blockedPatterns = [
      /\b(ignore|disregard|forget|override)\s+(previous|all|above)\s+(instructions?|rules?|constraints?)/gi,
      /\b(you\s+are\s+now|pretend\s+to\s+be|act\s+as|roleplay)/gi,
      /\[\s*SYSTEM\s*\]/gi,
      /<\s*script\s*>/gi,
      /\{[[\s\S]*?SYSTEM[\s\S]*?\}\}/gi
    ];

    this.maxInjectionAttempts = options.maxInjectionAttempts || 3;
    this.blockedUsers = new Map(); // userId -> attempt count
  }

  /**
   * 检测提示注入
   */
  check(userInput, userId) {
    const violations = [];

    for (const pattern of this.blockedPatterns) {
      const matches = userInput.match(pattern);
      if (matches) {
        violations.push({
          pattern: pattern.toString(),
          matches,
          severity: 'HIGH'
        });
      }
    }

    // 检查历史注入尝试
    const previousAttempts = this.blockedUsers.get(userId) || 0;
    if (violations.length > 0) {
      this.blockedUsers.set(userId, previousAttempts + 1);

      if (previousAttempts + 1 >= this.maxInjectionAttempts) {
        return {
          allowed: false,
          reason: 'INJECTION_DETECTED',
          message: '检测到多次尝试注入，系统已阻止此操作'
        };
      }
    } else {
      // 没有违规，清除计数
      this.blockedUsers.delete(userId);
    }

    return {
      allowed: true,
      violations,
      warning: violations.length > 0
        ? '输入可能包含恶意内容，已记录'
        : null
    };
  }

  /**
   * 安全构建 Prompt
   */
  buildSecurePrompt(systemPrompt, userInput, context = {}) {
    // 先检查用户输入
    const checkResult = this.check(userInput, context.userId);

    if (!checkResult.allowed) {
      throw new AppError('VALIDATION_INPUT_BLOCKED', checkResult.message, {
        httpStatus: 400
      });
    }

    // 安全地拼接 Prompt
    return {
      system: systemPrompt,
      user: userInput,
      context: {
        // 不包含敏感信息
        timestamp: Date.now(),
        sessionId: context.sessionId
      }
    };
  }
}
```

### 4.1 多层限流策略

```javascript
// middleware/rateLimit/multiLayerLimiter.js
/**
 * 多层限流器
 */
class MultiLayerRateLimiter {
  constructor(options = {}) {
    this.globalLimiter = new TokenBucketLimiter({
      capacity: options.globalCapacity || 1000,    // 全局限流
      refillRate: options.globalRefillRate || 100  // 每秒补充 100
    });

    this.userLimiter = new TokenBucketLimiter({
      capacity: options.userCapacity || 100,        // 用户级限流
      refillRate: options.userRefillRate || 10
    });

    this.endpointLimiters = new Map();
  }

  /**
   * 获取端点限流器
   */
  getEndpointLimiter(endpoint, options = {}) {
    if (!this.endpointLimiters.has(endpoint)) {
      this.endpointLimiters.set(endpoint, new TokenBucketLimiter({
        capacity: options.capacity || 50,
        refillRate: options.refillRate || 5
      }));
    }
    return this.endpointLimiters.get(endpoint);
  }

  /**
   * 检查限流
   */
  async check(userId, endpoint, tokens = 1) {
    const results = [];

    // 1. 全局限流检查
    const globalResult = await this.globalLimiter.tryConsume(tokens);
    results.push({ layer: 'global', ...globalResult });

    // 2. 用户级限流检查
    const userResult = await this.userLimiter.tryConsume(userId, tokens);
    results.push({ layer: 'user', key: userId, ...userResult });

    // 3. 端点限流检查
    const endpointLimiter = this.getEndpointLimiter(endpoint);
    const endpointResult = await endpointLimiter.tryConsume(userId, tokens);
    results.push({ layer: 'endpoint', key: endpoint, ...endpointResult });

    // 任一限流触发则拒绝
    const blocked = results.find(r => !r.allowed);

    if (blocked) {
      return {
        allowed: false,
        blockedBy: blocked.layer,
        retryAfter: blocked.retryAfter || 1,
        details: results
      };
    }

    return { allowed: true, details: results };
  }

  /**
   * 获取限流状态
   */
  async getStatus(userId, endpoint) {
    return {
      global: await this.globalLimiter.getStatus(),
      user: await this.userLimiter.getStatus(userId),
      endpoint: await this.getEndpointLimiter(endpoint).getStatus(userId)
    };
  }
}

/**
 * Token Bucket 限流器
 */
class TokenBucketLimiter {
  constructor(options) {
    this.capacity = options.capacity;
    this.refillRate = options.refillRate; // 每秒补充
    this.buckets = new Map(); // key -> { tokens, lastRefill }
  }

  async tryConsume(key, tokens = 1) {
    const bucket = this.getOrCreateBucket(key);

    if (bucket.tokens >= tokens) {
      bucket.tokens -= tokens;
      return { allowed: true, remaining: bucket.tokens };
    }

    // 计算需要等待的时间
    const deficit = tokens - bucket.tokens;
    const waitTime = deficit / this.refillRate;

    return {
      allowed: false,
      remaining: bucket.tokens,
      retryAfter: Math.ceil(waitTime)
    };
  }

  getOrCreateBucket(key) {
    if (!this.buckets.has(key)) {
      this.buckets.set(key, {
        tokens: this.capacity,
        lastRefill: Date.now()
      });
    }

    const bucket = this.buckets.get(key);

    // 补充 tokens
    const now = Date.now();
    const elapsed = (now - bucket.lastRefill) / 1000;
    const tokensToAdd = elapsed * this.refillRate;

    bucket.tokens = Math.min(this.capacity, bucket.tokens + tokensToAdd);
    bucket.lastRefill = now;

    return bucket;
  }

  async getStatus(key) {
    const bucket = this.getOrCreateBucket(key);
    return {
      capacity: this.capacity,
      remaining: bucket.tokens,
      refillRate: this.refillRate
    };
  }
}
```

### 4.2 限流中间件

```javascript
// middleware/rateLimit/rateLimitMiddleware.js
const multiLayerLimiter = new MultiLayerRateLimiter({
  globalCapacity: 1000,
  globalRefillRate: 100,
  userCapacity: 100,
  userRefillRate: 10
});

async function rateLimitMiddleware(req, res, next) {
  const userId = req.user?.id || req.ip;
  const endpoint = req.path;

  try {
    const result = await multiLayerLimiter.check(userId, endpoint);

    // 设置限流响应头
    res.set({
      'X-RateLimit-Limit': result.details.find(d => d.layer === 'endpoint')?.capacity || 0,
      'X-RateLimit-Remaining': result.details.find(d => d.layer === 'endpoint')?.remaining || 0
    });

    if (!result.allowed) {
      return res.status(429).json({
        success: false,
        code: '3005', // BUSINESS_LIMIT_EXCEEDED
        message: '请求过于频繁，请稍后重试',
        blockedBy: result.blockedBy,
        retryAfter: result.retryAfter
      });
    }

    next();
  } catch (error) {
    // 限流服务异常，放行请求但记录日志
    logger.warn('Rate limiter error', { error: error.message });
    next();
  }
}
```

### 5.1 敏感数据加密

```javascript
// services/security/encryption.js
const crypto = require('crypto');

const ALGORITHM = 'aes-256-gcm';
const KEY = crypto.scryptSync(process.env.ENCRYPTION_KEY, 'salt', 32);

/**
 * 加密敏感数据
 */
function encrypt(plaintext) {
  const iv = crypto.randomBytes(16);
  const cipher = crypto.createCipheriv(ALGORITHM, KEY, iv);

  let encrypted = cipher.update(plaintext, 'utf8', 'hex');
  encrypted += cipher.final('hex');

  const authTag = cipher.getAuthTag();

  return {
    encrypted,
    iv: iv.toString('hex'),
    authTag: authTag.toString('hex')
  };
}

/**
 * 解密敏感数据
 */
function decrypt(encryptedData) {
  const decipher = crypto.createDecipheriv(
    ALGORITHM,
    KEY,
    Buffer.from(encryptedData.iv, 'hex')
  );

  decipher.setAuthTag(Buffer.from(encryptedData.authTag, 'hex'));

  let decrypted = decipher.update(encryptedData.encrypted, 'hex', 'utf8');
  decrypted += decipher.final('utf8');

  return decrypted;
}

/**
 * 字段级别加密装饰器
 */
function encryptedField(target, fieldName) {
  const encryptedValue = Symbol(`${fieldName}_encrypted`);

  Object.defineProperty(target, fieldName, {
    get() {
      const enc = this[encryptedValue];
      return enc ? decrypt(enc) : null;
    },
    set(value) {
      this[encryptedValue] = value ? encrypt(value) : null;
    }
  });
}
```

```javascript
      'http://localhost:8080',
      'http://localhost:3000'
    ];

    // 允许没有 origin 的请求（如 Postman）
    if (!origin || allowedOrigins.includes(origin)) {
      callback(null, true);
    } else {
      callback(new Error('CORS not allowed'));
    }
  },
  credentials: true,
  methods: ['GET', 'POST', 'PUT', 'DELETE', 'OPTIONS'],
  allowedHeaders: [
    'Content-Type',
    'Authorization',
    'X-API-Key',
    'X-Trace-Id',
    'X-Request-ID'
  ],
  exposedHeaders: [
    'X-Trace-Id',
    'X-RateLimit-Limit',
    'X-RateLimit-Remaining'
  ],
  maxAge: 86400 // 预检请求缓存 24 小时
};
```

```javascript
// services/audit/AuditLogger.js
class AuditLogger {
  constructor(options = {}) {
    this.table = options.table || 'audit_logs';
    this.logger = options.logger;
  }

  async log(event, context) {
    const entry = {
      id: generateId(),
      event: event,
      category: AuditEvents[event]?.category || 'unknown',
      severity: AuditEvents[event]?.severity || 'INFO',
      userId: context.userId,
      ip: context.ip,
      userAgent: context.userAgent,
      resource: context.resource,
      resourceId: context.resourceId,
      metadata: context.metadata,
      timestamp: new Date()
    };

    // 异步写入，不阻塞主流程
    setImmediate(() => this.writeLog(entry));

    return entry.id;
  }

  async writeLog(entry) {
    try {
      await db.insert(this.table, entry);
    } catch (error) {
      this.logger.error('Audit log write failed', { entry, error: error.message });
    }
  }
}

// 使用示例
const auditLogger = new AuditLogger({ logger });

// 记录用户登录
await auditLogger.log('USER_LOGIN', {
  userId: user.id,
  ip: req.ip,
  metadata: { method: 'password' }
});

// 记录敏感操作
await auditLogger.log('API_KEY_CREATED', {
  userId: user.id,
  ip: req.ip,
  metadata: { keyId: apiKey.id, last4: apiKey.id.slice(-4) }
});
```
