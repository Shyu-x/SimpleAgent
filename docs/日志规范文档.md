## 一、日志设计原则

### 1.1 日志级别定义

| 级别 | 使用场景 | 示例 |
|------|---------|------|
| **ERROR** | 系统级错误，需要人工处理 | 数据库连接失败、第三方 API 完全不可用 |
| **WARN** | 潜在问题，可能需要关注 | 熔断器触发、限流触发、配置缺失 |
| **INFO** | 业务关键节点 | API 请求开始/结束、会话创建/关闭 |
| **DEBUG** | 开发调试信息 | 详细参数、中间状态 |
| **TRACE** | 最细粒度追踪 | 函数入口/出口、循环次数 |

### 1.2 日志格式规范

```
时间戳 [级别] [traceId] 消息 {结构化数据}
```

**示例**:
```
2026-03-21T10:30:00.123Z [INFO] [abc123def456] 用户登录成功 {userId: "user_xxx", ip: "192.168.1.1"}
2026-03-21T10:30:01.456Z [ERROR] [abc123def456] LLM API 调用失败 {model: "MiniMax-M2.7", error: "Connection timeout"}
```

### 1.3 JSON Lines 格式 (生产环境)

```json
{"timestamp":"2026-03-21T10:30:00.123Z","level":"INFO","traceId":"abc123def456","message":"用户登录成功","userId":"user_xxx","ip":"192.168.1.1"}
{"timestamp":"2026-03-21T10:30:01.456Z","level":"ERROR","traceId":"abc123def456","message":"LLM API 调用失败","model":"MiniMax-M2.7","error":"Connection timeout"}
```

## 三、日志埋点规范

### 3.1 API 请求埋点

```javascript
// middleware/requestLogger.js
function requestLogger(req, res, next) {
  const startTime = Date.now();
  const traceId = req.headers['x-trace-id'] || generateId();

  // 请求开始
  logger.info('Request started', {
    traceId,
    method: req.method,
    path: req.path,
    query: req.query,
    userId: req.user?.id
  });

  // 响应完成
  res.on('finish', () => {
    const duration = Date.now() - startTime;
    const level = res.statusCode >= 500 ? 'ERROR' :
                  res.statusCode >= 400 ? 'WARN' : 'INFO';

    logger.log(level, 'Request completed', {
      traceId,
      method: req.method,
      path: req.path,
      statusCode: res.statusCode,
      duration,
      userId: req.user?.id
    });
  });

  next();
}
```

### 3.2 服务层埋点

```javascript
// services/agentEngine.js
class AgentEngine {
  async run(userInput, options = {}) {
    const traceId = options.traceId || generateId();
    const startTime = Date.now();

    // 开始执行
    logger.info('Agent execution started', {
      traceId,
      agentId: this.agentId,
      inputLength: userInput.length,
      maxIterations: this.maxIterations
    });

    try {
      // 执行循环
      for (let i = 0; i < this.maxIterations; i++) {
        const iterationStart = Date.now();

        // Reason 阶段
        const thought = await this.think(userInput);
        logger.debug('Agent thought', {
          traceId,
          iteration: i + 1,
          phase: 'REASON',
          thinking: thought.substring(0, 200)
        });

        // Act 阶段
        if (thought.action) {
          const toolResult = await this.executeTool(thought.action);
          logger.debug('Tool executed', {
            traceId,
            iteration: i + 1,
            phase: 'ACT',
            tool: thought.action.tool,
            success: toolResult.success,
            duration: Date.now() - iterationStart
          });
        }

        // 检查是否完成
        if (this.isComplete()) {
          break;
        }
      }

      const totalDuration = Date.now() - startTime;
      logger.info('Agent execution completed', {
        traceId,
        iterations: this.state.iteration,
        duration: totalDuration
      });

      return this.state.result;

    } catch (error) {
      logger.error('Agent execution failed', {
        traceId,
        error: error.message,
        stack: error.stack,
        duration: Date.now() - startTime
      });
      throw error;
    }
  }
}
```

### 3.3 检索埋点

```javascript
// domain/search/SearchCoordinator.js
async search(query, options = {}) {
  const traceId = options.traceId || generateId();
  const startTime = Date.now();

  logger.info('Search started', {
    traceId,
    query: query.substring(0, 100),
    channels: this.channels.map(c => c.channelName),
    topK: options.topK
  });

  try {
    // 1. 执行通道
    const channelStart = Date.now();
    const channelResults = await Promise.all(
      this.channels.map(async (channel) => {
        const cStart = Date.now();
        try {
          const results = await channel.search(query, options);
          return {
            channel: channel.channelName,
            success: true,
            count: results.length,
            duration: Date.now() - cStart
          };
        } catch (error) {
          logger.warn('Channel search failed', {
            traceId,
            channel: channel.channelName,
            error: error.message
          });
          return {
            channel: channel.channelName,
            success: false,
            count: 0,
            duration: Date.now() - cStart,
            error: error.message
          };
        }
      })
    );

    // 2. 后处理
    const postProcessStart = Date.now();
    let merged = this.mergeResults(channelResults);
    const postProcessors = [];

    for (const processor of this.postProcessors) {
      const pStart = Date.now();
      const before = merged.length;
      merged = await processor.process(merged, query);
      postProcessors.push({
        name: processor.constructor.name,
        before,
        after: merged.length,
        duration: Date.now() - pStart
      });
    }

    const totalDuration = Date.now() - startTime;
    logger.info('Search completed', {
      traceId,
      channelResults,
      postProcessors,
      finalCount: merged.length,
      totalDuration
    });

    return merged;

  } catch (error) {
    logger.error('Search failed', {
      traceId,
      query: query.substring(0, 100),
      error: error.message,
      stack: error.stack
    });
    throw error;
  }
}
```

### 3.4 Pipeline 节点埋点

```javascript
// domain/rag/ingestion/IngestionPipeline.js
async process(document, options = {}) {
  const traceId = options.traceId || generateId();
  const context = { document, traceId };
  const nodeResults = [];

  logger.info('Pipeline started', {
    traceId,
    documentId: document.id,
    documentName: document.name,
    nodeCount: this.nodes.length
  });

  for (const node of this.nodes) {
    if (!node.shouldExecute(context)) {
      logger.debug('Node skipped', {
        traceId,
        node: node.nodeName,
        reason: 'shouldExecute returned false'
      });
      continue;
    }

    const nodeStart = Date.now();
    logger.info('Node started', {
      traceId,
      node: node.nodeName,
      order: node.order
    });

    try {
      const result = await node.execute(context);
      const duration = Date.now() - nodeStart;

      nodeResults.push({
        node: node.nodeName,
        success: true,
        duration,
        outputSize: result ? JSON.stringify(result).length : 0
      });

      logger.info('Node completed', {
        traceId,
        node: node.nodeName,
        duration,
        success: true
      });

      // 转换上下文
      context[node.outputKey] = result;

    } catch (error) {
      const duration = Date.now() - nodeStart;

      nodeResults.push({
        node: node.nodeName,
        success: false,
        duration,
        error: error.message
      });

      logger.error('Node failed', {
        traceId,
        node: node.nodeName,
        duration,
        error: error.message,
        stack: error.stack
      });

      if (!node.continueOnError) {
        throw error;
      }
    }
  }

  logger.info('Pipeline completed', {
    traceId,
    documentId: document.id,
    nodeResults,
    totalDuration: nodeResults.reduce((sum, r) => sum + r.duration, 0)
  });

  return context;
}
```

## 四、日志采集配置

### 4.1 Winston 配置

```javascript
// infra/logger/LoggerFactory.js
const winston = require('winston');
const { combine, timestamp, printf, json, colorize } = winston.format;

class LoggerFactory {
  static create(options = {}) {
    const { env = 'development', logDir = './logs' } = options;

    const formats = [
      timestamp({ format: 'YYYY-MM-DDTHH:mm:ss.SSSZ' }),
      env === 'development' ? colorize() : json()
    ];

    const consoleFormat = printf(({ timestamp, level, message, traceId, ...meta }) => {
      let log = `${timestamp} [${level}]`;
      if (traceId) log += ` [${traceId}]`;
      log += ` ${message}`;
      if (Object.keys(meta).length > 0) {
        log += ` ${JSON.stringify(meta)}`;
      }
      return log;
    });

    const transports = [
      // 控制台输出
      new winston.transports.Console({
        format: combine(...formats, consoleFormat)
      }),

      // 请求日志
      new winston.transports.File({
        filename: `${logDir}/request.log`,
        format: combine(...formats),
        maxsize: 100 * 1024 * 1024,  // 100MB
        maxFiles: 10
      }),

      // 错误日志
      new winston.transports.File({
        filename: `${logDir}/error.log`,
        level: 'error',
        format: combine(...formats),
        maxsize: 100 * 1024 * 1024,
        maxFiles: 10
      }),

      // 应用日志
      new winston.transports.File({
        filename: `${logDir}/application.log`,
        format: combine(...formats),
        maxsize: 100 * 1024 * 1024,
        maxFiles: 10
      })
    ];

    // 生产环境添加 JSON Lines 格式
    if (env === 'production') {
      transports.push(
        new winston.transports.File({
          filename: `${logDir}/application.jsonl`,
          format: combine(...formats, json()),
          maxsize: 100 * 1024 * 1024,
          maxFiles: 30
        })
      );
    }

    return winston.createLogger({
      level: env === 'production' ? 'INFO' : 'DEBUG',
      transports
    });
  }
}
```

## 五、日志使用规范

### 5.1 敏感信息处理

```javascript
// 日志脱敏
function sanitizeLogData(data) {
  const sensitiveFields = ['password', 'token', 'apiKey', 'secret', 'creditCard'];

  const sanitized = { ...data };

  for (const field of sensitiveFields) {
    if (sanitized[field]) {
      sanitized[field] = '***REDACTED***';
    }
  }

  return sanitized;
}

// 使用
logger.info('User login', sanitizeLogData({
  userId: 'user_xxx',
  password: 'user_password',  // 会被脱敏
  token: 'Bearer xxx'  // 会被脱敏
}));
```

### 5.2 日志级别选择指南

```javascript
// ERROR: 系统无法继续工作
logger.error('Database connection failed', { error: error.message, retryAttempts: 3 });

// WARN: 系统可以继续但需要关注
logger.warn('Rate limit approaching', { userId, currentCount, limit });
logger.warn('Circuit breaker opened', { model: 'MiniMax-M2.7', failureCount: 5 });

// INFO: 业务关键节点
logger.info('User action completed', { userId, action: 'create_document', duration });
logger.info('Batch job started', { jobId, totalItems: 1000 });

// DEBUG: 开发调试信息
logger.debug('Function parameters', { param1, param2, param3 });
logger.debug('Cache hit', { key: cacheKey });

// TRACE: 最细粒度追踪
logger.trace('Loop iteration', { iteration: i, total: 100 });
```

### 5.3 结构化日志最佳实践

```javascript
// ✅ 正确：使用结构化字段
logger.info('API request processed', {
  traceId,
  endpoint: '/api/chat/send',
```

## 六、日志分析示例

### 6.1 错误趋势分析

```bash
# 统计每分钟错误数
cat logs/error.log | jq -r '.timestamp[:16]' | sort | uniq -c | tail -20

# 统计错误类型分布
cat logs/error.log | jq -r '.errorName' | sort | uniq -c | sort -rn

# 查找特定 traceId 的所有日志
grep "abc123def456" logs/application.log | jq -s .
```

### 6.2 性能分析

```bash
# 统计 P50/P90/P99 响应时间
cat logs/request.log | jq -r '.responseTime' | \
  awk '{count[1]++; sum+=$1; arr[NR]=$1} END {
    sort(arr); n=NR;
    print "P50:", arr[int(n*0.5)];
    print "P90:", arr[int(n*0.9)];
    print "P99:", arr[int(n*0.99)];
    print "Avg:", sum/count[1];
  }'

# 统计慢请求
cat logs/request.log | jq -r 'select(.responseTime > 1000)' | wc -l
```

### 6.3 Agent 执行分析

```bash
# 统计 Agent 平均执行时间
cat logs/agent.log | jq -r 'select(.phase == "COMPLETE") | .duration' | \
  awk '{sum+=$1; count++} END {print "Avg:", sum/count}'

# 统计工具调用频率
cat logs/agent.log | jq -r '.toolCalls[].tool' | sort | uniq -c | sort -rn
```
