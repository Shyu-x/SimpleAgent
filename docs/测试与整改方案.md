
## 三、详细测试方案

### 3.1 熔断器深度测试

```javascript
// tests/stress/circuitBreakerStress.test.js

/**
 * 熔断器压力测试用例
 *
 * 测试场景：
 * 1. 故障注入 - 连续失败触发熔断
 * 2. 熔断恢复 - OPEN → HALF_OPEN → CLOSED
 * 3. 并发熔断 - 多线程同时触发
 * 4. 半开探测 - 成功/失败率对状态影响
 */

const scenarios = [
  {
    name: '故障注入测试',
    failureThreshold: 5,
    resetTimeout: 5000,
    probeTimeout: 2000,
    successThreshold: 2,
    injectedFailures: 10,
    expectedState: 'OPEN',
    validation: (breaker) => breaker.isOpen
  },
  {
    name: '熔断恢复测试',
    openDuration: 6000,  // 超过 resetTimeout
    expectedState: 'HALF_OPEN',
    validation: (breaker) => breaker.isHalfOpen
  },
  {
    name: '半开成功恢复测试',
    probeSuccesses: 3,
    expectedState: 'CLOSED',
    validation: (breaker) => breaker.isClosed
  },
  {
    name: '并发故障触发测试',
    concurrentRequests: 100,
    expected: '至少 1 个请求被熔断拒绝',
    validation: (results) => results.rejected > 0
  }
];
```

**验收标准**:
- [ ] 5 次连续失败后进入 OPEN 状态
- [ ] 熔断期间拒绝所有请求
- [ ] resetTimeout 后进入 HALF_OPEN
- [ ] 探测成功后正确恢复到 CLOSED
- [ ] 并发请求无竞态条件
