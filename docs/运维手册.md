## 三、部署方案

### 3.1 Docker 部署（推荐）

#### 目录结构

```
/opt/ai-chat/
├── docker-compose.yml
├── .env
├── backend/
│   └── data/           # 后端数据
├── redis/
│   └── data/           # Redis 数据
├── postgres/
│   └── data/           # PostgreSQL 数据
└── logs/               # 日志目录
    ├── backend/
    ├── frontend/
    └── nginx/
```

#### docker-compose.yml

```yaml
version: '3.8'

services:
  frontend:
    image: ai-chat-frontend:latest
    container_name: ai-chat-frontend
    ports:
      - "8080:3000"
    environment:
      - NODE_ENV=production
      - NEXT_PUBLIC_BACKEND_URL=http://backend:30000
    depends_on:
      - backend
    networks:
      - ai-chat-network
    restart: unless-stopped
    volumes:
      - ./logs/frontend:/var/log/nginx

  backend:
    image: ai-chat-backend:latest
    container_name: ai-chat-backend
    ports:
      - "30000:30000"
    environment:
      - NODE_ENV=production
      - MINIMAX_API_KEY=${MINIMAX_API_KEY}
      - MINIMAX_BASE_URL=${MINIMAX_BASE_URL:-https://api.minimaxi.com/anthropic}
      - REDIS_HOST=redis
      - REDIS_PORT=6379
      - DB_HOST=postgres
      - DB_PORT=5432
      - DB_NAME=ai_chat
      - DB_USER=ai_chat
      - DB_PASSWORD=${DB_PASSWORD}
    depends_on:
      - redis
      - postgres
    networks:
      - ai-chat-network
    restart: unless-stopped
    volumes:
      - ./backend/data:/app/data
      - ./logs/backend:/app/logs

  redis:
    image: redis:7-alpine
    container_name: ai-chat-redis
    ports:
      - "6379:6379"
    command: redis-server --appendonly yes --maxmemory 512mb --maxmemory-policy allkeys-lru
    volumes:
      - ./redis/data:/data
    networks:
      - ai-chat-network
    restart: unless-stopped

  postgres:
    image: postgres:15-alpine
    container_name: ai-chat-postgres
    ports:
      - "5432:5432"
    environment:
      - POSTGRES_DB=ai_chat
      - POSTGRES_USER=ai_chat
      - POSTGRES_PASSWORD=${DB_PASSWORD}
    volumes:
      - ./postgres/data:/var/lib/postgresql/data
    networks:
      - ai-chat-network
    restart: unless-stopped

  nginx:
    image: nginx:alpine
    container_name: ai-chat-nginx
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./nginx/ssl:/etc/nginx/ssl:ro
    depends_on:
      - frontend
      - backend
    networks:
      - ai-chat-network
    restart: unless-stopped

networks:
  ai-chat-network:
    driver: bridge
```

### 3.2 手动部署

#### 后端部署

```bash
# 1. 创建用户
sudo useradd -m -s /bin/bash app
sudo mkdir -p /opt/ai-chat/backend
sudo chown app:app /opt/ai-chat/backend

# 2. 安装 Node.js
curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -
sudo apt-get install -y nodejs

# 3. 部署代码
sudo -u app git clone <repo-url> /opt/ai-chat/backend
cd /opt/ai-chat/backend
sudo -u app npm install --production

# 4. 配置环境变量
sudo -u app vim /opt/ai-chat/backend/.env

# 5. 创建 systemd 服务
sudo tee /etc/systemd/system/ai-chat-backend.service << EOF
[Unit]
Description=AI Chat Backend
After=network.target

[Service]
Type=simple
User=app
WorkingDirectory=/opt/ai-chat/backend
ExecStart=/usr/bin/node src/index.js
Restart=always
RestartSec=10
Environment=NODE_ENV=production

[Install]
WantedBy=multi-user.target
EOF

# 6. 启动服务
sudo systemctl daemon-reload
sudo systemctl enable ai-chat-backend
sudo systemctl start ai-chat-backend
```

### 3.3 Nginx 配置

```nginx
# /etc/nginx/nginx.conf
worker_processes auto;
error_log /var/log/nginx/error.log warn;
pid /var/run/nginx.pid;

events {
    worker_connections 1024;
    use epoll;
    multi_accept on;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    # 日志格式
    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                    '$status $body_bytes_sent "$http_referer" '
                    '"$http_user_agent" "$http_x_forwarded_for" '
                    'rt=$request_time uct="$upstream_connect_time" '
                    'uht="$upstream_header_time" urt="$upstream_response_time"';

    access_log /var/log/nginx/access.log main;

    # 性能优化
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    types_hash_max_size 2048;

    # Gzip 压缩
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_comp_level 6;
    gzip_types text/plain text/css text/xml application/json application/javascript application/rss+xml application/atom+xml image/svg+xml;

    # 上游服务
    upstream frontend {
        server 127.0.0.1:8080;
        keepalive 32;
    }

    upstream backend {
        server 127.0.0.1:30000;
        keepalive 32;
    }

    server {
        listen 80;
        server_name _;

        # 重定向到 HTTPS（生产环境）
        # return 301 https://$host$request_uri;

        # 前端静态文件
        location / {
            proxy_pass http://frontend;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection 'upgrade';
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_cache_bypass $http_upgrade;
        }

        # API 代理
        location /api/ {
            proxy_pass http://backend;
            proxy_http_version 1.1;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        # SSE 流式响应
        location /stream {
            proxy_pass http://backend;
            proxy_http_version 1.1;
            proxy_set_header Connection '';
            proxy_set_header Cache-Control 'no-cache';
            proxy_set_header X-Accel-Buffering 'no-cache';
            proxy_set_header X-Real-IP $remote_addr;
        }
    }

    # HTTPS 服务器（生产环境）
```

### 4.3 告警规则

```yaml
# rules/backend.yml
groups:
  - name: ai-chat-backend
    rules:
      # API 响应时间告警
      - alert: APIHighLatency
        expr: histogram_quantile(0.99, rate(http_request_duration_seconds_bucket[5m])) > 2
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "API 响应时间过高"
          description: "P99 响应时间超过 2 秒，当前值: {{ $value }}s"

      # API 错误率告警
      - alert: APIHighErrorRate
        expr: rate(http_requests_total{status=~"5.."}[5m]) / rate(http_requests_total[5m]) > 0.01
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "API 错误率过高"
          description: "错误率超过 1%，当前值: {{ $value | humanizePercentage }}"

      # 服务不可用告警
      - alert: BackendDown
        expr: up{job="ai-chat-backend"} == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "后端服务不可用"
          description: "后端服务已停止响应"

      # Agent 执行超时告警
      - alert: AgentExecutionTimeout
        expr: rate(agent_execution_timeout_total[5m]) > 0.1
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Agent 执行超时"
```

## 六、故障排查

### 6.1 故障分类

| 等级 | 定义 | 响应时间 | 解决时间 |
|------|------|---------|---------|
| P0 | 核心功能不可用 | 5 分钟 | 1 小时 |
| P1 | 主要功能受损 | 15 分钟 | 4 小时 |
| P2 | 次要功能异常 | 1 小时 | 24 小时 |
| P3 | 轻微问题 | 4 小时 | 72 小时 |

### 6.2 故障排查流程

```
1. 确认故障现象
   ├── 查看监控告警
   ├── 查看用户反馈
   └── 确认影响范围

2. 快速止血
   ├── 切换备用服务
   ├── 降级非核心功能
   └── 启用限流保护

3. 定位根因
   ├── 查看应用日志
   ├── 查看系统日志
   ├── 查看监控指标
   └── 分析链路追踪

4. 修复问题
   ├── 代码修复
   ├── 配置调整
   └── 服务重启

5. 验证恢复
   ├── 功能测试
   ├── 监控确认
   └── 通知相关方
```

### 6.3 常见故障处理

#### 故障 1: 后端服务无响应

**排查步骤**:

```bash
# 1. 检查进程状态
docker ps | grep backend
systemctl status ai-chat-backend

# 2. 检查资源使用
docker stats ai-chat-backend --no-stream
free -h
df -h

# 3. 检查网络连接
netstat -an | grep 30000
ss -s

# 4. 查看日志
docker logs ai-chat-backend --tail 100
journalctl -u ai-chat-backend -n 100

# 5. 检查上游依赖
ping redis
nc -zv redis 6379
ping postgres
nc -zv postgres 5432
```

**可能原因**:
- 内存不足导致 OOM
- 上游服务连接超时
- 代码死锁或死循环

**解决方案**:

```bash
# 重启服务
docker restart ai-chat-backend
systemctl restart ai-chat-backend

# 增加内存限制
docker update --memory=2g ai-chat-backend

# 检查 OOM
dmesg | grep -i "out of memory"
```

#### 故障 2: SSE 连接断开

**排查步骤**:

```bash
# 1. 检查 Nginx 配置
nginx -t

# 2. 检查 Nginx 日志
tail -f /var/log/nginx/error.log
tail -f /var/log/nginx/access.log

# 3. 检查后端 SSE 状态
curl -v http://localhost:30000/api/health

# 4. 检查连接数
netstat -an | grep ESTABLISHED | wc -l
```

**可能原因**:
- Nginx 超时配置过短
- 后端服务重启
- 网络不稳定

**解决方案**:

```nginx
# Nginx 配置调整
proxy_read_timeout 86400;
proxy_send_timeout 86400;
proxy_connect_timeout 86400;
```

#### 故障 3: Token 消耗异常

**排查步骤**:

```bash
# 1. 查看 Token 消耗日志
grep "token_consumed" logs/backend/app.log | tail -100

# 2. 分析消耗来源
# 检查是否为异常请求

# 3. 检查是否有恶意调用

```

### 7.2 备份脚本

```bash
#!/bin/bash
# backup.sh - 数据备份脚本

set -e

# 配置
BACKUP_DIR="/opt/ai-chat/backups"
DATE=$(date +%Y%m%d_%H%M%S)
RETENTION_DAYS=30

# 创建备份目录
mkdir -p ${BACKUP_DIR}

# PostgreSQL 备份
echo "Starting PostgreSQL backup..."
PGPASSWORD=${DB_PASSWORD} pg_dump -h localhost -U ai_chat -d ai_chat | gzip > ${BACKUP_DIR}/postgres_${DATE}.sql.gz

# Redis 备份
echo "Starting Redis backup..."
redis-cli SAVE
cp /var/lib/redis/dump.rdb ${BACKUP_DIR}/redis_${DATE}.rdb

# 清理旧备份
echo "Cleaning up old backups..."
find ${BACKUP_DIR} -name "*.sql.gz" -mtime +${RETENTION_DAYS} -delete
find ${BACKUP_DIR} -name "*.rdb" -mtime +${RETENTION_DAYS} -delete

# 上传到对象存储
echo "Uploading to object storage..."
# rclone copy ${BACKUP_DIR}/postgres_${DATE}.sql.gz s3:bucket/ai-chat-backups/

echo "Backup completed: ${DATE}"
```

### 7.3 恢复流程

```bash
#!/bin/bash
# restore.sh - 数据恢复脚本

set -e

# 参数
BACKUP_FILE=$1
if [ -z "$BACKUP_FILE" ]; then
  echo "Usage: $0 <backup_file>"
  exit 1
fi

echo "Starting restore from: ${BACKUP_FILE}"

# 停止服务
docker-compose down

# 恢复 PostgreSQL
echo "Restoring PostgreSQL..."
gunzip -c ${BACKUP_FILE} | PGPASSWORD=${DB_PASSWORD} psql -h localhost -U ai_chat -d ai_chat

# 恢复 Redis
echo "Restoring Redis..."
redis-cli SHUTDOWN NOSAVE || true
cp ${BACKUP_DIR}/redis_*.rdb /var/lib/redis/dump.rdb
chown redis:redis /var/lib/redis/dump.rdb
redis-server --daemonize yes

# 启动服务
docker-compose up -d

echo "Restore completed successfully"
```
