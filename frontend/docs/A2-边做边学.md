## 实战 1：添加新的聊天功能

### 1.1 修改 Zustand Store
**文件**: `frontend/src/store/chatStore.ts`

**任务**: 添加"消息已读"功能

```typescript
// 1. 添加新状态
interface ChatState {
  messages: Message[];
  readMessages: Set<string>;  // 新增：已读消息 ID
  // ...
}

// 2. 添加新方法
const useChatStore = create<ChatState>((set, get) => ({
  // ...
  readMessages: new Set(),

  // 标记消息为已读
  markAsRead: (messageId: string) => {
    set((state) => {
      const newSet = new Set(state.readMessages);
      newSet.add(messageId);
      return { readMessages: newSet };
    });
  },

  // 获取未读消息数
  getUnreadCount: () => {
    const state = get();
    return state.messages.filter(m => !state.readMessages.has(m.id)).length;
  }
}));
```

### 1.2 在组件中使用
**文件**: `frontend/src/components/ChatArea.tsx`

```tsx
function ChatArea() {
  const { messages, markAsRead, getUnreadCount } = useChatStore();

  // 当消息显示时标记为已读
  useEffect(() => {
    messages.forEach(msg => {
      markAsRead(msg.id);
    });
  }, [messages, markAsRead]);

  return (
    <div className="chat-area">
      {/* 未读消息数 */}
      {getUnreadCount() > 0 && (
        <div className="unread-badge">{getUnreadCount()} 条未读</div>
      )}

      {/* 消息列表 */}
      {messages.map(msg => (
        <MessageBubble key={msg.id} message={msg} />
      ))}
    </div>
  );
}
```

## 实战 2：添加新的 UI 状态

### 2.1 创建 UI Store
**文件**: `frontend/src/stores/uiStore.ts` (如果不存在)

```typescript
import { create } from 'zustand';

interface UIState {
  // 侧边栏
  sidebarOpen: boolean;
  toggleSidebar: () => void;

  // 主题
  theme: 'light' | 'dark';
  toggleTheme: () => void;

  // 模态框
  activeModal: string | null;
  openModal: (id: string) => void;
  closeModal: () => void;
}

export const useUIStore = create<UIState>((set) => ({
  sidebarOpen: true,
  toggleSidebar: () => set((state) => ({ sidebarOpen: !state.sidebarOpen })),

  theme: 'light',
  toggleTheme: () => set((state) => ({
    theme: state.theme === 'light' ? 'dark' : 'light'
  })),

  activeModal: null,
  openModal: (id) => set({ activeModal: id }),
  closeModal: () => set({ activeModal: null }),
}));
```

### 2.2 在页面中使用
**文件**: `frontend/src/app/page.tsx`

```tsx
'use client';

import { useUIStore } from '@/stores/uiStore';

export default function Home() {
  const { sidebarOpen, toggleSidebar, theme } = useUIStore();

  return (
    <div className={`app ${theme}`}>
      {/* 侧边栏切换按钮 */}
      <button onClick={toggleSidebar}>
        {sidebarOpen ? '隐藏侧边栏' : '显示侧边栏'}
      </button>

      {/* 主题切换 */}
      <button onClick={() => useUIStore.getState().toggleTheme()}>
        当前主题: {theme}
      </button>

      {/* 主聊天区域 */}
      <ChatArea />
    </div>
  );
}
```

## 实战 3：添加自定义 Hook

### 3.1 创建 Hook
**文件**: `frontend/src/hooks/useLocalStorage.ts`

```typescript
import { useState, useEffect } from 'react';

// 通用的 localStorage Hook
export function useLocalStorage<T>(key: string, initialValue: T) {
  // 获取初始值
  const [storedValue, setStoredValue] = useState<T>(() => {
    if (typeof window === 'undefined') {
      return initialValue;
    }
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.error(error);
      return initialValue;
    }
  });

  // 更新值
  const setValue = (value: T | ((val: T) => T)) => {
    try {
      const valueToStore = value instanceof Function ? value(storedValue) : value;
      setStoredValue(valueToStore);
      if (typeof window !== 'undefined') {
        window.localStorage.setItem(key, JSON.stringify(valueToStore));
      }
    } catch (error) {
      console.error(error);
    }
  };

  return [storedValue, setValue] as const;
}
```

### 3.2 使用 Hook
```tsx
// 在组件中使用
function Settings() {
  const [username, setUsername] = useLocalStorage('username', '');

  return (
    <div>
      <input
        value={username}
        onChange={(e) => setUsername(e.target.value)}
        placeholder="输入用户名"
      />
      <p>已保存: {username}</p>
    </div>
  );
}
```

## 实战 4：调用后端 API

### 4.1 配置 API 客户端
**文件**: `frontend/src/lib/apiClient.ts`

```typescript
const API_BASE = process.env.NEXT_PUBLIC_BACKEND_URL || 'http://localhost:30000';

export async function fetchAIResponse(message: string, sessionId?: string) {
  const response = await fetch(`${API_BASE}/api/chat`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({ message, sessionId }),
  });

  if (!response.ok) {
    throw new Error(`API error: ${response.status}`);
  }

  return response;
}
```

### 4.2 SSE 流式接收
**文件**: `frontend/src/hooks/useStreamChat.ts`

```typescript
import { useState, useCallback } from 'react';

export function useStreamChat() {
  const [isStreaming, setIsStreaming] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const sendMessage = useCallback(async (message: string) => {
    setIsStreaming(true);
    setError(null);

    try {
      const response = await fetch(`${API_BASE}/api/chat`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ message }),
      });

      if (!response.ok) throw new Error('请求失败');

      // 读取 SSE 流
      const reader = response.body?.getReader();
      const decoder = new TextDecoder();

      if (!reader) throw new Error('无法读取响应流');

      while (true) {
        const { done, value } = await reader.read();
        if (done) break;

        const chunk = decoder.decode(value);
        console.log('收到数据:', chunk);
      }
    } catch (err) {
      setError(err instanceof Error ? err.message : '未知错误');
    } finally {
      setIsStreaming(false);
    }
  }, []);

  return { sendMessage, isStreaming, error };
}
```

## 常见问题排查

### 问题：组件不更新
1. 检查 Zustand 是否用了正确的 hook（`useStore` 而不是直接 import）
2. 检查是否正确调用了 `set()`
3. 使用 React DevTools 查看状态

### 问题：SSE 不工作
1. 检查后端是否运行
2. 检查 CORS 配置
3. 检查 API URL 是否正确

### 问题：样式丢失
1. 检查 Tailwind 配置
2. 检查 className 拼写
3. 确保组件是 `'use client'`
