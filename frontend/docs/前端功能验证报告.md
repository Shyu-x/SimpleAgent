### 2.2 SSE任务状态Hook (useTaskStatus.ts)

**文件**: `frontend/src/hooks/useTaskStatus.ts` (520行)

**功能说明**:
- 提供任务状态管理、SSE连接、自动重连等功能
- 支持事件回调（任务创建/开始/进度/完成/失败等）
- 自动重连机制（默认5次重试，间隔3秒）

**配置选项**:
```typescript
interface UseTaskStatusOptions {
  collaborationId?: string;       // 协作会话 ID
  taskId?: string;               // 任务 ID
  autoConnect?: boolean;         // 是否自动连接
  sseEndpoint?: string;          // 自定义 SSE 端点
  maxReconnectAttempts?: number; // 最大重连次数（默认5）
  reconnectInterval?: number;    // 重连间隔（默认3000ms）
  connectionTimeout?: number;    // 连接超时（默认30000ms）
  autoReconnect?: boolean;      // 是否自动重连

  // 事件回调
  onTaskCreated?: (task: Task) => void;
  onTaskStarted?: (taskId: string) => void;
  onTaskProgress?: (taskId: string, progress: number, stage?: string) => void;
  onTaskStage?: (taskId: string, stage: string) => void;
  onTaskComplete?: (taskId: string, result: unknown) => void;
  onTaskFailed?: (taskId: string, error: ErrorInfo) => void;
  onTaskCancelled?: (taskId: string, reason?: string) => void;
  onStreamChunk?: (taskId: string, chunk: StreamChunk) => void;
  onLog?: (taskId: string, level: string, message: string) => void;
  onToolCall?: (taskId: string, toolName: string, toolArgs?: object) => void;
  onConfirmation?: (request: ConfirmationRequest) => void;

  // 连接状态回调
  onConnected?: () => void;
  onDisconnected?: () => void;
  onConnectionError?: (error: Error) => void;
}
```

**返回接口**:
```typescript
interface UseTaskStatusReturn {
  isConnected: boolean;              // 是否已连接
  connectionStatus: ConnectionStatus; // 连接状态
  lastError: ErrorInfo | null;        // 最后错误信息

  getTask: (taskId: string) => Task | undefined;
  getAllTasks: () => Task[];
  getActiveTasks: () => Task[];
  getStreamBuffer: (taskId: string) => StreamChunk[];

  connect: () => void;
  disconnect: () => void;
  refreshTask: (taskId: string) => Promise<void>;
  clearTask: (taskId: string) => void;
  clearAllTasks: () => void;
}
```

**SSE端点**:
- 默认: `/api/a2a/collaboration/${collaborationId}/subscribe`
- 备选: `/api/a2a/collaboration/${taskId}/subscribe`
- 自定义: 通过 `sseEndpoint` 参数指定

### 2.3 任务事件处理器单例 (taskEventHandler.ts)

**文件**: `frontend/src/lib/taskEventHandler.ts` (485行)

**功能说明**:
- 任务事件处理器单例，支持自定义处理器注册
- 批量事件处理、优先级排序
- SSE事件解析为标准化事件

**事件类型**:
```typescript
type TaskEventType =
  | 'task_created'    // 任务创建
  | 'task_started'    // 任务开始
  | 'task_progress'   // 任务进度
  | 'task_stage'      // 任务阶段变化
  | 'task_complete'   // 任务完成
  | 'task_failed'     // 任务失败
  | 'task_cancelled'  // 任务取消
  | 'stream_chunk'    // 流数据块
  | 'log'             // 日志
  | 'tool_call'       // 工具调用
  | 'confirmation';   // 确认请求
```

**核心API**:
```typescript
// 获取单例实例
getTaskEventHandler(): TaskEventProcessor

// 注册事件处理器
register(
  handler: TaskEventHandler,
  eventTypes?: TaskEventType[],  // 要处理的事件类型
  taskIds?: string[],             // 要处理的任务ID
  priority?: number               // 优先级（数字越大越先执行）
): string  // 返回处理器注册ID

// 注销处理器
unregister(handlerId: string): boolean

// 发布事件（异步队列处理）
emit(event: TaskEvent): void

// 同步发布事件（立即处理）
emitSync(event: TaskEvent): void

// 批量发布事件
emitBatch(events: TaskEvent[]): void

// SSE事件解析
parseSSEEvent(eventType: string, data: unknown): TaskEvent | null
```

**使用示例**:
```typescript
const handler = getTaskEventHandler();

// 注册全局日志处理器
handler.register((event) => {
  if (event.type === 'log') {
    console.log(`[${event.level}] ${event.message}`);
  }
}, ['log'], undefined, 100);

// 注册特定任务的事件处理器
handler.register(
  (event) => { /* 处理事件 */ },
  undefined,  // 处理所有事件类型
  ['task-123', 'task-456'],  // 只处理这两个任务
  50  // 较低优先级
);
```
